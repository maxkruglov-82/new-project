findRangeOrigin(array<float> savedOpens, array<float> savedCloses, array<int> savedTimes, int wantedSide, int fromTime, int beforeFirst) =>
    float foundPrice = na
    int foundTime = na
    int bestSpan = -1
    int count = array.size(savedTimes)
    if wantedSide == 1 and count > 1
        for anchorIndex = 0 to count - 2
            int anchorTime = array.get(savedTimes, anchorIndex)
            float anchorOpen = array.get(savedOpens, anchorIndex)
            float anchorClose = array.get(savedCloses, anchorIndex)
            float anchorPrice = wantedSide == 1 ? math.min(anchorOpen, anchorClose) : math.max(anchorOpen, anchorClose)
            bool localAnchor = anchorIndex == 0
            if anchorIndex > 0
                float previousOpen = array.get(savedOpens, anchorIndex - 1)
                float previousClose = array.get(savedCloses, anchorIndex - 1)
                localAnchor := wantedSide == 1 ? anchorPrice < math.min(previousOpen, previousClose) : anchorPrice > math.max(previousOpen, previousClose)
            float nextClose = array.get(savedCloses, anchorIndex + 1)
            bool startsRetreat = array.get(savedTimes, anchorIndex + 1) - anchorTime == 300000 and (wantedSide == 1 ? nextClose >= anchorPrice : nextClose <= anchorPrice)
            if anchorTime >= fromTime and anchorTime < beforeFirst and localAnchor and startsRetreat
                float retreatExtreme = anchorPrice
                int retreatTime = anchorTime
                int retreatSpan = 1
                bool following = true
                for rightIndex = anchorIndex + 1 to count - 1
                    if following
                        int rightTime = array.get(savedTimes, rightIndex)
                        if rightTime >= beforeFirst or rightTime - anchorTime != (rightIndex - anchorIndex) * 300000
                            following := false
                        else
                            float rightOpen = array.get(savedOpens, rightIndex)
                            float rightClose = array.get(savedCloses, rightIndex)
                            float rightExtreme = wantedSide == 1 ? math.max(rightOpen, rightClose) : math.min(rightOpen, rightClose)
                            bool reachesExtreme = wantedSide == 1 ? rightExtreme >= retreatExtreme : rightExtreme <= retreatExtreme
                            if reachesExtreme
                                retreatExtreme := rightExtreme
                                retreatTime := rightTime
                                retreatSpan := rightIndex - anchorIndex + 1
                            if wantedSide == 1 ? rightClose < anchorPrice : rightClose > anchorPrice
                                following := false
                bool betterPrice = na(foundPrice) or (wantedSide == 1 ? retreatExtreme > foundPrice : retreatExtreme < foundPrice)
                if retreatExtreme > anchorPrice and retreatSpan >= 5 and (retreatSpan > bestSpan or (retreatSpan == bestSpan and betterPrice))
                    foundPrice := retreatExtreme
                    foundTime := retreatTime
                    bestSpan := retreatSpan
    if wantedSide == -1 and count > 1
        for scanIndex = 0 to count - 2
            int scanTime = array.get(savedTimes, scanIndex)
            if scanTime >= fromTime and scanTime < beforeFirst
                float ceiling = math.max(array.get(savedOpens, scanIndex), array.get(savedCloses, scanIndex))
                float candidate = na
                int candidateTime = na
                int span = 1
                bool scanning = true
                for nextIndex = scanIndex + 1 to count - 1
                    if scanning
                        int nextTime = array.get(savedTimes, nextIndex)
                        float nextOpen = array.get(savedOpens, nextIndex)
                        float nextClose = array.get(savedCloses, nextIndex)
                        if nextTime >= beforeFirst or nextTime - scanTime != (nextIndex - scanIndex) * 300000 or math.max(nextOpen, nextClose) > ceiling
                            scanning := false
                        else
                            span := span + 1
                            float nextLow = math.min(nextOpen, nextClose)
                            if na(candidate) or nextLow <= candidate
                                candidate := nextLow
                                candidateTime := nextTime
                bool betterLow = na(foundPrice) or candidate < foundPrice or (candidate == foundPrice and candidateTime > foundTime)
                if span >= 5 and not na(candidate) and (span > bestSpan or (span == bestSpan and betterLow))
                    foundPrice := candidate
                    foundTime := candidateTime
                    bestSpan := span
    [foundPrice, foundTime]
retreatStopHasMinimum(array<float> savedOpens, array<float> savedCloses, array<int> savedTimes, int anchorTime, float stopPrice, bool lowerStop, int minimumBars) =>
    bool started = false
    bool finished = false
    bool qualified = false
    int span = 0
    int previousTime = na
    int count = array.size(savedTimes)
    if count > 0 and not na(anchorTime) and not na(stopPrice)
        for si = 0 to count - 1
            int st = array.get(savedTimes, si)
            if not finished and st >= anchorTime
                if not started
                    if st == anchorTime
                        started := true
                        span := 1
                        previousTime := st
                    else
                        finished := true
                else if st - previousTime != 300000
                    finished := true
                else
                    span := span + 1
                    previousTime := st
                    float so = array.get(savedOpens, si)
                    float sc = array.get(savedCloses, si)
                    float edge = lowerStop ? math.min(so, sc) : math.max(so, sc)
                    if edge == stopPrice
                        qualified := span >= minimumBars
                        finished := true
    qualified

drawColoredRuns(array<chart.point> points, array<polyline> drawings, array<polyline> history, color ink) =>
    array<chart.point> run = array.new<chart.point>()
    int previousCandle = na
    int count = array.size(points)
    if count >= 5
        for candleIndex = 0 to int(count / 5) - 1
            int base = candleIndex * 5
            chart.point firstPoint = array.get(points, base)
            int candleStart = firstPoint.time
            if not na(previousCandle) and candleStart - previousCandle != 300000
                saveColoredRun(run, drawings, history, ink)
                array.clear(run)
            for component = 0 to 3
                array.push(run, array.get(points, base + component))
            previousCandle := candleStart
        if array.size(run) > 0
            saveColoredRun(run, drawings, history, ink)
    array.size(drawings)
rebuildColoredDay(array<chart.point> redPoints, array<chart.point> greenPoints, array<float> savedOpens, array<float> savedCloses, array<int> savedTimes, int zeroTime, float zeroPrice, float firstPrice, int originBreakTime, int fourthBreakTime, float thirdPrice, float fourthPrice, bool fifthReady, int fifthTime, float fifthPrice, int fifthSide, float offset, int endTime) =>
    array.clear(redPoints)
    array.clear(greenPoints)
    int initialSide = firstPrice > zeroPrice ? 1 : -1
    int trend = initialSide
    float floorPrice = math.min(zeroPrice, firstPrice)
    float ceilingPrice = math.max(zeroPrice, firstPrice)
    bool originApplied = false
    bool fourthApplied = false
    int result = 0
    int count = array.size(savedTimes)
    if count > 0
        for candleIndex = 0 to count - 1
            int candleTime = array.get(savedTimes, candleIndex)
            float candleOpen = array.get(savedOpens, candleIndex)
            float candleClose = array.get(savedCloses, candleIndex)
            if candleTime >= zeroTime and (na(endTime) or candleTime <= endTime)
                if not originApplied and not na(originBreakTime) and candleTime >= originBreakTime
                    originApplied := true
                    trend := -initialSide
                    floorPrice := trend == 1 ? firstPrice : math.min(candleOpen, candleClose)
                    ceilingPrice := trend == 1 ? math.max(candleOpen, candleClose) : firstPrice
                if originApplied
                    if not na(fourthBreakTime) and candleTime >= fourthBreakTime and not fourthApplied
                        fourthApplied := true
                        if fourthPrice > thirdPrice
                            floorPrice := thirdPrice
                            ceilingPrice := math.max(ceilingPrice, math.max(candleOpen, candleClose))
                        else
                            ceilingPrice := thirdPrice
                            floorPrice := math.min(floorPrice, math.min(candleOpen, candleClose))
                    if trend == 1 and floorPrice - candleClose >= offset - 0.000000001
                        trend := -1
                    else if trend == -1 and candleClose - ceilingPrice >= offset - 0.000000001
                        trend := 1
                    if trend == 1
                        ceilingPrice := math.max(ceilingPrice, math.max(candleOpen, candleClose))
                    else
                        floorPrice := math.min(floorPrice, math.min(candleOpen, candleClose))
                    if fifthReady and candleTime >= fifthTime
                        if fifthSide == 1
                            ceilingPrice := fifthPrice
                        else
                            floorPrice := fifthPrice
                if trend == 1
                    appendColoredCandle(greenPoints, candleTime, candleOpen, candleClose)
                else
                    appendColoredCandle(redPoints, candleTime, candleOpen, candleClose)
                result := trend
    result

recordMarketSwing(array<int> times, array<float> prices, array<bool> highs, array<int> days, int atTime, float price, bool isHigh, int dayKey) =>
    int count = array.size(times)
    bool sameDay = count > 0 ? array.get(days, count - 1) == dayKey : false
    bool sameType = sameDay ? array.get(highs, count - 1) == isHigh : false
    if sameType
        float oldPrice = array.get(prices, count - 1)
        if isHigh ? price >= oldPrice : price <= oldPrice
            array.set(times, count - 1, atTime)
            array.set(prices, count - 1, price)
    else
        array.push(times, atTime)
        array.push(prices, price)
        array.push(highs, isHigh)
        array.push(days, dayKey)
    while array.size(times) > 120
        array.shift(times)
        array.shift(prices)
        array.shift(highs)
        array.shift(days)
