// Повторный поиск после определения границ H1: внешний экстремум
// не должен блокировать допустимое начало движения внутри этих границ.
findBoundedOrigin(array<float> savedOpens, array<float> savedCloses, array<int> savedTimes, int wantedSide, float upperBound, float lowerBound, int beforeFirst) =>
    float foundPrice = na
    int foundTime = na
    int legSide = wantedSide
    float primaryPrice = na
    float primaryOpen = na
    int primaryTime = na
    int primaryIndex = na
    float oppositePrice = na
    float oppositeOpen = na
    int oppositeTime = na
    int oppositeIndex = na
    int count = array.size(savedTimes)
    if count > 0
        for scanIndex = 0 to count - 1
            float scanOpen = array.get(savedOpens, scanIndex)
            float scanClose = array.get(savedCloses, scanIndex)
            int scanTime = array.get(savedTimes, scanIndex)
            float scanHigh = math.max(scanOpen, scanClose)
            float scanLow = math.min(scanOpen, scanClose)
            // Pine v5 может вычислять обе стороны and: array.get(-1) недопустим.
            if scanIndex > 0
                int previousSavedTime = array.get(savedTimes, scanIndex - 1)
                if scanTime - previousSavedTime != 300000
                    primaryPrice := na
                    oppositePrice := na
                    legSide := wantedSide
            float legPrice = legSide == 1 ? scanHigh : scanLow
            bool legAllowed = legSide != wantedSide or (wantedSide == 1 ? legPrice <= upperBound : legPrice >= lowerBound)
            bool localSeed = false
            if scanIndex > 0 and scanIndex < count - 1
                float leftOpen = array.get(savedOpens, scanIndex - 1)
                float leftClose = array.get(savedCloses, scanIndex - 1)
                float rightOpen = array.get(savedOpens, scanIndex + 1)
                float rightClose = array.get(savedCloses, scanIndex + 1)
                localSeed := legSide == 1 ? legPrice >= math.max(leftOpen, leftClose) and legPrice > math.max(rightOpen, rightClose) : legPrice <= math.min(leftOpen, leftClose) and legPrice < math.min(rightOpen, rightClose)
            bool extendsLeg = legAllowed and (na(primaryPrice) ? localSeed : (legSide == 1 ? legPrice > primaryPrice : legPrice < primaryPrice))
            if extendsLeg
                primaryPrice := legPrice
                primaryOpen := scanOpen
                primaryTime := scanTime
                primaryIndex := scanIndex
                oppositePrice := na
            float otherPrice = legSide == 1 ? scanLow : scanHigh
            bool otherAllowed = -legSide != wantedSide or (wantedSide == 1 ? otherPrice <= upperBound : otherPrice >= lowerBound)
            if not na(primaryPrice) and otherAllowed and (na(oppositePrice) or (legSide == 1 ? otherPrice < oppositePrice : otherPrice > oppositePrice))
                oppositePrice := otherPrice
                oppositeOpen := scanOpen
                oppositeTime := scanTime
                oppositeIndex := scanIndex
            bool closedRetreat = not na(primaryPrice) and scanIndex - primaryIndex + 1 >= 4 and scanTime - primaryTime == (scanIndex - primaryIndex) * 300000 and (legSide == 1 ? scanClose < primaryOpen : scanClose > primaryOpen)
            if closedRetreat and not na(oppositePrice)
                if legSide == wantedSide and primaryTime < beforeFirst
                    foundPrice := primaryPrice
                    foundTime := primaryTime
                legSide := -legSide
                primaryPrice := oppositePrice
                primaryOpen := oppositeOpen
                primaryTime := oppositeTime
                primaryIndex := oppositeIndex
                oppositePrice := na
                for innerIndex = primaryIndex to scanIndex
                    float innerOpen = array.get(savedOpens, innerIndex)
                    float innerClose = array.get(savedCloses, innerIndex)
                    float innerPrice = legSide == 1 ? math.min(innerOpen, innerClose) : math.max(innerOpen, innerClose)
                    bool innerAllowed = -legSide != wantedSide or (wantedSide == 1 ? innerPrice <= upperBound : innerPrice >= lowerBound)
                    if innerAllowed and (na(oppositePrice) or (legSide == 1 ? innerPrice < oppositePrice : innerPrice > oppositePrice))
                        oppositePrice := innerPrice
                        oppositeOpen := innerOpen
                        oppositeTime := array.get(savedTimes, innerIndex)
                        oppositeIndex := innerIndex
    [foundPrice, foundTime]

        if crossedSecond or crossedFirst
            confirmedThirdPrice := crossedSecond ? pullbackExtreme : (structureSide == -1 ? bodyLow : bodyHigh)
            int thirdPointTime = crossedSecond ? pullbackExtremeTime : time
            label thirdLabel = label.new(x=thirdPointTime, y=confirmedThirdPrice, text="3", xloc=xloc.bar_time, yloc=yloc.price, style=structureSide == -1 ? label.style_label_up : label.style_label_down, color=color.new(color.red, 100), textcolor=color.red, size=size.large)
            if crossedSecond
                // 4 — новый экстремум свечи, закрепившейся за 2; 3 — край отката перед ней.
                confirmedFourthPrice := structureSide == -1 ? bodyHigh : bodyLow
                int fourthPointTime = time
                label fourthLabel = label.new(x=fourthPointTime, y=confirmedFourthPrice, text="4", xloc=xloc.bar_time, yloc=yloc.price, style=structureSide == -1 ? label.style_label_down : label.style_label_up, color=color.new(color.red, 100), textcolor=color.red, size=size.large)
                movingFourthLabel := fourthLabel
                bool fourthOnCurrent = true
                fourthAnchorOpen := fourthOnCurrent ? open : open[1]
                fourthExtremeBar := fourthOnCurrent ? bar_index : bar_index - 1
                fourthRetreatCloses := 1
                fourthLocked := false
                array.push(firstPointLabels, fourthLabel)
            movingThirdLabel := thirdLabel
            thirdOppositeBody := crossedSecond ? pullbackOpen : open
            thirdExtremeBar := crossedSecond ? pullbackExtremeBar : thirdPointTime == time ? bar_index : thirdPointTime == time[1] ? bar_index - 1 : bar_index
            thirdRetreatCloses := 1
            thirdLocked := crossedSecond
            array.push(firstPointLabels, thirdLabel)
            structureCompleted := true
            if positionState != 2 and dailyPositionEntries == 0
                newPositionPlan := true
                positionIsSecond := false
                firstPositionStop := crossedSecond ? confirmedThirdPrice : confirmedSecondPrice
                firstPositionSide := crossedSecond ? -structureSide : structureSide
                firstPositionUsesFourth := crossedSecond
                positionSide := crossedSecond ? -structureSide : structureSide
                positionUsesFourth := crossedSecond
                positionStop := crossedSecond ? confirmedThirdPrice : confirmedSecondPrice
                positionTarget := crossedSecond ? confirmedFourthPrice : confirmedThirdPrice
                positionThreshold := (positionStop + positionTarget) / 2.0
                positionState := 1
                positionPlanDay := rangeDayKey
                positionSetupBar := bar_index
                positionEntryBar := na
                positionEntry := na
                positionRR := na
                positionRiskBox := na
                positionRewardBox := na
                positionInfo := na
        else
            float retreatPrice = structureSide == -1 ? bodyLow : bodyHigh
            if na(pullbackExtreme) or (structureSide == -1 ? retreatPrice < pullbackExtreme : retreatPrice > pullbackExtreme)
                pullbackExtreme := retreatPrice
                pullbackExtremeTime := time
                pullbackOpen := open
                pullbackOppositeBody := structureSide == -1 ? bodyHigh : bodyLow
                pullbackExtremeBar := bar_index
    while array.size(firstPointLabels) > 400
        label.delete(array.shift(firstPointLabels))


// Подтверждённая 3 сопровождает продолжение до отдельного четырёхсвечного отката.
if barstate.isconfirmed and firstPointWindow and m5Chart and structureDay == rangeDayKey and structureCompleted and not thirdLocked and not na(movingThirdLabel)
    float nextThirdPrice = structureSide == -1 ? bodyLow : bodyHigh
    bool extendsThird = structureSide == -1 ? nextThirdPrice < confirmedThirdPrice : nextThirdPrice > confirmedThirdPrice
    if extendsThird
        confirmedThirdPrice := nextThirdPrice
        thirdOppositeBody := open
        thirdExtremeBar := bar_index
        thirdRetreatCloses := 1
        label.set_xy(movingThirdLabel, time, confirmedThirdPrice)
    else if bar_index > thirdExtremeBar and not na(thirdOppositeBody)
        bool outsideThirdBody = structureSide == -1 ? close > thirdOppositeBody : close < thirdOppositeBody
        thirdRetreatCloses := time - time[1] == 300000 ? thirdRetreatCloses + 1 : 1
        if thirdRetreatCloses >= 4 and outsideThirdBody
            thirdLocked := true


// 4 движется до отката: минимум четыре M5, включая свечу экстремума,
// последняя закрывается за open первой в обратную сторону.
if barstate.isconfirmed and firstPointWindow and m5Chart and structureDay == rangeDayKey and not na(movingFourthLabel) and not fourthLocked
    float fourthCandidate = structureSide == -1 ? bodyHigh : bodyLow
    bool extendsFourth = structureSide == -1 ? fourthCandidate > confirmedFourthPrice : fourthCandidate < confirmedFourthPrice
    if extendsFourth
        confirmedFourthPrice := fourthCandidate
        fourthAnchorOpen := open
        fourthExtremeBar := bar_index
        fourthRetreatCloses := 1
        label.set_xy(movingFourthLabel, time, confirmedFourthPrice)
    else if bar_index > fourthExtremeBar
        bool fourthRetreatClose = structureSide == -1 ? close < fourthAnchorOpen : close > fourthAnchorOpen
        fourthRetreatCloses := time - time[1] == 300000 ? fourthRetreatCloses + 1 : 1
        if fourthRetreatCloses >= 4 and fourthRetreatClose
            fourthLocked := true


// Следующий экстремум 5 после противоположного пробоя за стопом первой позиции.
if barstate.isconfirmed and firstPointWindow and m5Chart and positionIsSecond and not na(fifthLabel) and not fifthLocked
    float fifthCandidate = positionSide == 1 ? bodyHigh : bodyLow
    bool extendsFifth = positionSide == 1 ? fifthCandidate > confirmedFifthPrice : fifthCandidate < confirmedFifthPrice
    if extendsFifth
        confirmedFifthPrice := fifthCandidate
        fifthAnchorOpen := open
        fifthExtremeBar := bar_index
        fifthRetreatBars := 1
        label.set_xy(fifthLabel, time, confirmedFifthPrice)
    else if bar_index > fifthExtremeBar
        fifthRetreatBars := time - time[1] == 300000 ? fifthRetreatBars + 1 : 1
        if fifthRetreatBars >= 4 and (positionSide == 1 ? close < fifthAnchorOpen : close > fifthAnchorOpen)
            fifthLocked := true

// Инструмент позиции: стоп без отступа, закрытие M5 с фактическим RR >= 1.
// State: 0 none, 1 waiting, 2 entered, 3 TP, 4 SL, 5 cancelled/expired, 6 ambiguous.
int positionExitEvent = 0
bool positionEntryEvent = false
if barstate.isconfirmed and m5Chart
    if positionState == 1
        if positionPlanDay != rangeDayKey or not firstPointWindow
            positionState := 5
        else
            float currentTarget = positionIsSecond ? confirmedFifthPrice : positionUsesFourth ? confirmedFourthPrice : confirmedThirdPrice
            positionTarget := positionSide == 1 ? math.max(positionTarget, currentTarget) : math.min(positionTarget, currentTarget)
            positionThreshold := (positionStop + positionTarget) / 2.0
            bool stopBeforeEntry = positionSide == 1 ? bodyLow <= positionStop : bodyHigh >= positionStop
            if bar_index > positionSetupBar and stopBeforeEntry
                positionState := 5
            else if bar_index > positionSetupBar
                float riskAtClose = positionSide == 1 ? close - positionStop : positionStop - close
                float rewardAtClose = positionSide == 1 ? positionTarget - close : close - positionTarget
                if dailyPositionEntries < 2 and (not positionIsSecond or firstStopped) and riskAtClose > 0 and rewardAtClose > 0 and rewardAtClose + 0.000000001 >= riskAtClose
                    dailyPositionEntries := dailyPositionEntries + 1
                    positionEntry := close
                    positionRR := rewardAtClose / riskAtClose
                    positionEntryBar := bar_index
                    positionEntryEvent := true
                    positionState := 2
    // Внутрисвечное движение ДО входа по close не является результатом сделки.
    if positionState == 2 and bar_index > positionEntryBar
        bool hitStop = positionSide == 1 ? bodyLow <= positionStop : bodyHigh >= positionStop
        bool hitTarget = positionSide == 1 ? bodyHigh >= positionTarget : bodyLow <= positionTarget
        if hitStop or hitTarget
            positionExitEvent := hitStop and hitTarget ? 2 : hitStop ? -1 : 1
            positionState := hitStop and hitTarget ? 6 : hitStop ? 4 : 3
            if positionState == 4 and dailyPositionEntries == 1 and positionPlanDay == rangeDayKey
                firstStopped := true
                firstStopBar := bar_index
    // Только стоп первой исполненной позиции разрешает вторую, противоположную.
    bool reverseBreak = firstPositionSide == -1 ? close - firstPositionStop >= firstPointOffset - 0.000000001 : firstPositionStop - close >= firstPointOffset - 0.000000001
    if firstPointWindow and positionPlanDay == rangeDayKey and firstStopped and dailyPositionEntries == 1 and positionState == 4 and not positionIsSecond and reverseBreak
        // Завершаем рисунок первой позиции до создания нового инструмента.
        if not na(positionInfo)
            label.set_text(positionInfo, (firstPositionSide == 1 ? "LONG" : "SHORT") + " · Стоп\nВход: " + str.tostring(positionEntry, "#.#####") + " · Стоп: " + str.tostring(positionStop, "#.#####"))
            box.set_right(positionRiskBox, time_close)
            box.set_right(positionRewardBox, time_close)
        positionSide := -firstPositionSide
        positionIsSecond := true
        positionStop := firstPositionUsesFourth ? confirmedFourthPrice : confirmedThirdPrice
        confirmedFifthPrice := positionSide == 1 ? bodyHigh : bodyLow
        positionTarget := confirmedFifthPrice
        positionThreshold := (positionStop + positionTarget) / 2.0
        positionState := 1
        positionSetupBar := bar_index
        positionEntryBar := na
        positionEntry := na
        positionRR := na
        positionRiskBox := na
        positionRewardBox := na
        positionInfo := na
        newPositionPlan := true
        fifthAnchorOpen := open
        fifthExtremeBar := bar_index
        fifthRetreatBars := 1
        fifthLocked := false
        fifthLabel := label.new(x=time, y=confirmedFifthPrice, text=firstPositionUsesFourth ? "5" : "4", xloc=xloc.bar_time, yloc=yloc.price, style=positionSide == 1 ? label.style_label_down : label.style_label_up, color=color.new(color.red, 100), textcolor=color.red, size=size.large)
        array.push(firstPointLabels, fifthLabel)
        if array.size(firstPointLabels) > 400
            label.delete(array.shift(firstPointLabels))
    if newPositionPlan
        positionRiskBox := box.new(left=time, top=math.max(positionStop, positionThreshold), right=time_close, bottom=math.min(positionStop, positionThreshold), xloc=xloc.bar_time, border_color=color.red, bgcolor=color.new(color.red, 85))
        positionRewardBox := box.new(left=time, top=math.max(positionTarget, positionThreshold), right=time_close, bottom=math.min(positionTarget, positionThreshold), xloc=xloc.bar_time, border_color=color.teal, bgcolor=color.new(color.teal, 85))
        positionInfo := label.new(x=time, y=positionThreshold, text="Ожидание RR ≥ 1:1", xloc=xloc.bar_time, yloc=yloc.price, style=label.style_label_left, color=color.new(color.white, 15), textcolor=color.black, size=size.small)
        array.push(positionBoxes, positionRiskBox)
        array.push(positionBoxes, positionRewardBox)
        array.push(positionLabels, positionInfo)
        while array.size(positionBoxes) > 100
            box.delete(array.shift(positionBoxes))
        while array.size(positionLabels) > 50
            label.delete(array.shift(positionLabels))
    if not na(positionInfo) and (positionState == 1 or positionState == 2 or positionEntryEvent or positionState != positionState[1])
        float drawnEntry = na(positionEntry) ? positionThreshold : positionEntry
        box.set_top(positionRiskBox, math.max(positionStop, drawnEntry))
        box.set_bottom(positionRiskBox, math.min(positionStop, drawnEntry))
        box.set_top(positionRewardBox, math.max(positionTarget, drawnEntry))
        box.set_bottom(positionRewardBox, math.min(positionTarget, drawnEntry))
        box.set_right(positionRiskBox, time_close)
        box.set_right(positionRewardBox, time_close)
        label.set_xy(positionInfo, time, drawnEntry)
        string sideText = positionSide == 1 ? "LONG" : "SHORT"
        string statusText = positionState == 1 ? "Ожидание RR ≥ 1:1" : positionState == 2 ? "Вход по close" : positionState == 3 ? "Тейк" : positionState == 4 ? "Стоп" : positionState == 6 ? "TP/SL: порядок неизвестен" : "Без входа · отменено"
        label.set_text(positionInfo, sideText + " · " + statusText + "\n" + (na(positionEntry) ? "Граница RR1: " : "Вход: ") + str.tostring(drawnEntry, "#.#####") + "\nСтоп: " + str.tostring(positionStop, "#.#####") + " · Цель: " + str.tostring(positionTarget, "#.#####") + (na(positionRR) ? "" : "\nRR 1:" + str.tostring(positionRR, "#.##")))
plot(positionEntryEvent ? positionSide : 0, "Вход LONG/SHORT", display=display.none)
plot(positionState, "Состояние позиции", display=display.data_window)
plot(positionExitEvent, "Выход TP 1 / SL -1 / неизвестно 2", display=display.data_window)
plot(positionStop, "Стоп позиции", display=display.data_window)
plot(positionTarget, "Цель позиции", display=display.data_window)
plot(positionThreshold, "Граница RR 1:1", display=display.data_window)
plot(positionEntry, "Цена входа по close", display=display.data_window)
plot(positionRR, "RR входа", display=display.data_window)
plot(dailyPositionEntries, "Входов за день", display=display.data_window)
