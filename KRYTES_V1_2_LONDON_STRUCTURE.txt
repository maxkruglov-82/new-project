v12LocalAnchor(array<float> opens, array<float> closes, array<int> times, int fromTime, int untilTime, int side) =>
    float result = na
    int atTime = na
    int count = array.size(times)
    if count > 0
        for i = 0 to count - 1
            int t = array.get(times,i)
            if t > fromTime and t < untilTime
                float p = side==1?math.min(array.get(opens,i),array.get(closes,i)):math.max(array.get(opens,i),array.get(closes,i))
                if na(result) or (side==1?p<result:p>result)
                    result := p
                    atTime := t
    [result,atTime]

v12Advance(int side, int stage, float a, int at, float b, int bt, float c, int ct, float price, int t, bool isHigh) =>
    int next = stage
    float nextA = a
    int nextAT = at
    float nextB = b
    int nextBT = bt
    float nextC = c
    int nextCT = ct
    bool firstType = side==1?isHigh:not isHigh
    if stage==0 and firstType
        next := 1
        nextA := price
        nextAT := t
    else if stage==1 and t>=at
        if firstType and (side==1?price>a:price<a)
            nextA := price
            nextAT := t
        else if not firstType and t>at and (side==1?price<a:price>a)
            next := 2
            nextB := price
            nextBT := t
    else if stage==2 and t>bt
        if firstType and (side==1?price>a:price<a)
            next := 3
            nextC := price
            nextCT := t
        else if not firstType and (side==1?price<b:price>b)
            nextB := price
            nextBT := t
    [next,nextA,nextAT,nextB,nextBT,nextC,nextCT]

// Закрепы за всеми сформированными точками: два close M5 подряд.
// Каждое закрытие минимум на offset за одним и тем же уровнем.
// За сформированной 1 тоже два; только первый выход за час — одно закрытие.
twoM5Break(float level, bool upward, float offset, bool adjacentSameDay) =>
    bool currentBeyond = upward ? close - level >= offset - 0.000000001 : level - close >= offset - 0.000000001
    bool previousBeyond = upward ? close[1] - level >= offset - 0.000000001 : level - close[1] >= offset - 0.000000001
    adjacentSameDay and currentBeyond and previousBeyond

// Закреп мог состояться до подтверждения третьего свинга; повторно его не требуем.
retryBreakAfter(array<float> closes, array<int> times, float level, bool upward, float offset, int afterTime) =>
    bool confirmed = false
    bool previousBeyond = false
    int previousTime = na
    if array.size(times) > 0
        for i = 0 to array.size(times) - 1
            int atTime = array.get(times, i)
            if atTime >= afterTime
                float atClose = array.get(closes, i)
                bool beyond = upward ? atClose - level >= offset - 0.000000001 : level - atClose >= offset - 0.000000001
                if beyond and previousBeyond and atTime - previousTime == 300000
                    confirmed := true
                previousBeyond := beyond
                previousTime := atTime
    confirmed

// Точка 0 — экстремум собственного отката от противоположного экстремума.
// От 13:00 строго до 1; минимум пять свечей от начала отката включительно.
findRangeOrigin(array<float> savedOpens, array<float> savedCloses, array<int> savedTimes, int wantedSide, int fromTime, int beforeFirst) =>
    float foundPrice = na
    int foundTime = na
    int bestSpan = -1
    int count = array.size(savedTimes)
    if wantedSide == 1 and count > 1
        for anchorIndex = 0 to count - 2
            int anchorTime = array.get(savedTimes, anchorIndex)
            float anchorOpen = array.get(savedOpens, anchorIndex)
            float anchorClose = array.get(savedCloses, anchorIndex)
            float anchorPrice = wantedSide == 1 ? math.min(anchorOpen, anchorClose) : math.max(anchorOpen, anchorClose)
            bool localAnchor = anchorIndex == 0
            if anchorIndex > 0
                float previousOpen = array.get(savedOpens, anchorIndex - 1)
                float previousClose = array.get(savedCloses, anchorIndex - 1)
                localAnchor := wantedSide == 1 ? anchorPrice < math.min(previousOpen, previousClose) : anchorPrice > math.max(previousOpen, previousClose)
            float nextClose = array.get(savedCloses, anchorIndex + 1)
            bool startsRetreat = array.get(savedTimes, anchorIndex + 1) - anchorTime == 300000 and (wantedSide == 1 ? nextClose >= anchorPrice : nextClose <= anchorPrice)
            if anchorTime >= fromTime and anchorTime < beforeFirst and localAnchor and startsRetreat
                float retreatExtreme = anchorPrice
                int retreatTime = anchorTime
                int retreatSpan = 1
                bool following = true
                for rightIndex = anchorIndex + 1 to count - 1
                    if following
                        int rightTime = array.get(savedTimes, rightIndex)
                        if rightTime >= beforeFirst or rightTime - anchorTime != (rightIndex - anchorIndex) * 300000
                            following := false
                        else
                            float rightOpen = array.get(savedOpens, rightIndex)
                            float rightClose = array.get(savedCloses, rightIndex)
                            float rightExtreme = wantedSide == 1 ? math.max(rightOpen, rightClose) : math.min(rightOpen, rightClose)
                            bool reachesExtreme = wantedSide == 1 ? rightExtreme >= retreatExtreme : rightExtreme <= retreatExtreme
                            if reachesExtreme
                                retreatExtreme := rightExtreme
                                retreatTime := rightTime
                                retreatSpan := rightIndex - anchorIndex + 1
                            // Открытие пробойной свечи ещё может повторить экстремум.
                            if wantedSide == 1 ? rightClose < anchorPrice : rightClose > anchorPrice
                                following := false
                bool betterPrice = na(foundPrice) or (wantedSide == 1 ? retreatExtreme > foundPrice : retreatExtreme < foundPrice)
                if retreatExtreme > anchorPrice and retreatSpan >= 5 and (retreatSpan > bestSpan or (retreatSpan == bestSpan and betterPrice))
                    foundPrice := retreatExtreme
                    foundTime := retreatTime
                    bestSpan := retreatSpan
    // LOW 0: проверяем КАЖДУЮ свечу от 13:00, а не только локальные хаи.
    // Участок заканчивается перед первым обновлением максимума тела.
    // Начальная свеча входит в длину (минимум 5), но её low не кандидат.
    if wantedSide == -1 and count > 1
        for scanIndex = 0 to count - 2
            int scanTime = array.get(savedTimes, scanIndex)
            if scanTime >= fromTime and scanTime < beforeFirst
                float ceiling = math.max(array.get(savedOpens, scanIndex), array.get(savedCloses, scanIndex))
                float candidate = na
                int candidateTime = na
                int span = 1
                bool scanning = true
                for nextIndex = scanIndex + 1 to count - 1
                    if scanning
                        int nextTime = array.get(savedTimes, nextIndex)
                        float nextOpen = array.get(savedOpens, nextIndex)
                        float nextClose = array.get(savedCloses, nextIndex)
                        if nextTime >= beforeFirst or nextTime - scanTime != (nextIndex - scanIndex) * 300000 or math.max(nextOpen, nextClose) > ceiling
                            scanning := false
                        else
                            span := span + 1
                            float nextLow = math.min(nextOpen, nextClose)
                            if na(candidate) or nextLow <= candidate
                                candidate := nextLow
                                candidateTime := nextTime
                // Сначала число свечей; при равной длине — более низкий low.
                // При равном low выбираем последнее повторение.
                // 16.10: от 13:00 максимум 15 свечей выбирает LOW 106,08 / 14:35.
                bool betterLow = na(foundPrice) or candidate < foundPrice or (candidate == foundPrice and candidateTime > foundTime)
                if span >= 5 and not na(candidate) and (span > bestSpan or (span == bestSpan and betterLow))
                    foundPrice := candidate
                    foundTime := candidateTime
                    bestSpan := span
    [foundPrice, foundTime]

// Опора отката допустима, только если до ПЕРВОГО появления её тела
// прошло минимум minimumBars M5, включая свечу исходного экстремума.
// Повтор того же low/high на открытии следующей свечи длину не увеличивает.
retreatStopHasMinimum(array<float> savedOpens, array<float> savedCloses, array<int> savedTimes, int anchorTime, float stopPrice, bool lowerStop, int minimumBars) =>
    bool started = false
    bool finished = false
    bool qualified = false
    int span = 0
    int previousTime = na
    int count = array.size(savedTimes)
    if count > 0 and not na(anchorTime) and not na(stopPrice)
        for si = 0 to count - 1
            int st = array.get(savedTimes, si)
            if not finished and st >= anchorTime
                if not started
                    if st == anchorTime
                        started := true
                        span := 1
                        previousTime := st
                    else
                        finished := true
                else if st - previousTime != 300000
                    finished := true
                else
                    span := span + 1
                    previousTime := st
                    float so = array.get(savedOpens, si)
                    float sc = array.get(savedCloses, si)
                    float edge = lowerStop ? math.min(so, sc) : math.max(so, sc)
                    if edge == stopPrice
                        qualified := span >= minimumBars
                        finished := true
    qualified

appendColoredCandle(array<chart.point> points, int candleTime, float candleOpen, float candleClose) =>
    array.push(points, chart.point.from_time(candleTime - 60000, candleOpen))
    array.push(points, chart.point.from_time(candleTime, candleOpen))
    array.push(points, chart.point.from_time(candleTime, candleClose))
    array.push(points, chart.point.from_time(candleTime + 60000, candleClose))
    array.push(points, chart.point.from_time(candleTime + 120000, na))

// Отдельная ломаная на непрерывный участок одного цвета.
// Не полагаемся на na: терминал может соединять соседние конечные точки.
saveColoredRun(array<chart.point> run, array<polyline> drawings, array<polyline> history, color ink) =>
    while array.size(history) + array.size(drawings) >= 94
        if array.size(history) > 0
            polyline.delete(array.shift(history))
        else
            polyline.delete(array.shift(drawings))
    array.push(drawings, polyline.new(run, xloc=xloc.bar_time, line_color=ink, line_width=2))

drawColoredRuns(array<chart.point> points, array<polyline> drawings, array<polyline> history, color ink) =>
    array<chart.point> run = array.new<chart.point>()
    int previousCandle = na
    int count = array.size(points)
    if count >= 5
        for candleIndex = 0 to int(count / 5) - 1
            int base = candleIndex * 5
            chart.point firstPoint = array.get(points, base)
            int candleStart = firstPoint.time
            if not na(previousCandle) and candleStart - previousCandle != 300000
                saveColoredRun(run, drawings, history, ink)
                array.clear(run)
            for component = 0 to 3
                array.push(run, array.get(points, base + component))
            previousCandle := candleStart
        if array.size(run) > 0
            saveColoredRun(run, drawings, history, ink)
    array.size(drawings)

// Начальная смена цвета — сразу после двух закрытий за исходной 0.
// После неё сохраняем подтверждение последующих встречных троек.
rebuildColoredDay(array<chart.point> redPoints, array<chart.point> greenPoints, array<float> savedOpens, array<float> savedCloses, array<int> savedTimes, int zeroTime, float zeroPrice, float firstPrice, int firstTime, float offset, int endTime) =>
    array.clear(redPoints)
    array.clear(greenPoints)
    int trend = firstPrice > zeroPrice ? 1 : -1
    int initialTrend = trend
    bool initialZeroBroken = false
    int zeroBreakColorTime = na
    float latestHigh = trend == -1 ? zeroPrice : na
    float latestLow = trend == 1 ? zeroPrice : na
    int reverseStage = 0
    float reverseFirst = na
    int reverseFirstTime = na
    float reverseBase = na
    float reverseMiddle = na
    int reverseMiddleTime = na
    int result = 0
    int count = array.size(savedTimes)
    array<int> candleTrends = array.new_int(count, 0)
    if count > 0
        for candleIndex = 0 to count - 1
            int candleTime = array.get(savedTimes, candleIndex)
            float candleOpen = array.get(savedOpens, candleIndex)
            float candleClose = array.get(savedCloses, candleIndex)
            if candleTime >= zeroTime and (na(endTime) or candleTime <= endTime)
                if not initialZeroBroken and candleIndex > 0 and candleTime > firstTime
                    int priorTime = array.get(savedTimes, candleIndex - 1)
                    float priorClose = array.get(savedCloses, candleIndex - 1)
                    bool beyondZeroNow = initialTrend == 1 ? zeroPrice - candleClose >= offset - 0.000000001 : candleClose - zeroPrice >= offset - 0.000000001
                    bool beyondZeroBefore = initialTrend == 1 ? zeroPrice - priorClose >= offset - 0.000000001 : priorClose - zeroPrice >= offset - 0.000000001
                    if priorTime > firstTime and candleTime - priorTime == 300000 and beyondZeroNow and beyondZeroBefore
                        trend := -initialTrend
                        initialZeroBroken := true
                        zeroBreakColorTime := candleTime
                        reverseStage := 0
                if candleIndex >= 5
                    int pivotIndex = candleIndex - 4
                    int pivotTime = array.get(savedTimes, pivotIndex)
                    bool contiguousPivot = candleTime - array.get(savedTimes, pivotIndex - 1) == 1500000
                    if pivotTime >= zeroTime and contiguousPivot
                        float pivotHigh = math.max(array.get(savedOpens, pivotIndex), array.get(savedCloses, pivotIndex))
                        float pivotLow = math.min(array.get(savedOpens, pivotIndex), array.get(savedCloses, pivotIndex))
                        float leftHigh = math.max(array.get(savedOpens, pivotIndex - 1), array.get(savedCloses, pivotIndex - 1))
                        float leftLow = math.min(array.get(savedOpens, pivotIndex - 1), array.get(savedCloses, pivotIndex - 1))
                        bool highReady = pivotHigh >= leftHigh
                        bool lowReady = pivotLow <= leftLow
                        for rightIndex = pivotIndex + 1 to candleIndex
                            float rightHigh = math.max(array.get(savedOpens, rightIndex), array.get(savedCloses, rightIndex))
                            float rightLow = math.min(array.get(savedOpens, rightIndex), array.get(savedCloses, rightIndex))
                            highReady := highReady and pivotHigh >= rightHigh
                            lowReady := lowReady and pivotLow <= rightLow
                        if highReady != lowReady
                            bool pivotIsHigh = highReady
                            float pivotPrice = highReady ? pivotHigh : pivotLow
                            bool wantUp = trend == -1
                            bool firstType = pivotIsHigh == wantUp
                            float priorSame = pivotIsHigh ? latestHigh : latestLow
                            float priorOpposite = pivotIsHigh ? latestLow : latestHigh
                            if pivotTime > firstTime
                                if firstType
                                    bool newFirst = not na(priorSame) and not na(priorOpposite) and (wantUp ? pivotPrice > priorSame : pivotPrice < priorSame)
                                    if reverseStage == 0 and newFirst
                                        reverseFirst := pivotPrice
                                        reverseFirstTime := pivotTime
                                        reverseBase := priorOpposite
                                        reverseStage := 1
                                    else if reverseStage == 1 and (wantUp ? pivotPrice > reverseFirst : pivotPrice < reverseFirst)
                                        reverseFirst := pivotPrice
                                        reverseFirstTime := pivotTime
                                    else if reverseStage >= 2 and pivotTime > reverseMiddleTime and (wantUp ? pivotPrice > reverseFirst : pivotPrice < reverseFirst)
                                        reverseStage := 3
                                else if reverseStage >= 1 and pivotTime > reverseFirstTime
                                    bool validMiddle = wantUp ? pivotPrice > reverseBase and pivotPrice < reverseFirst : pivotPrice < reverseBase and pivotPrice > reverseFirst
                                    if validMiddle
                                        reverseMiddle := pivotPrice
                                        reverseMiddleTime := pivotTime
                                        reverseStage := 2
                                    else
                                        reverseStage := 0
                            if pivotIsHigh
                                latestHigh := pivotPrice
                            else
                                latestLow := pivotPrice
                if reverseStage == 3
                    bool wantUp = trend == -1
                    bool anchorBreak = false
                    bool middleBreak = false
                    bool previousAnchor = false
                    bool previousMiddle = false
                    int proofPreviousTime = na
                    for proofIndex = 0 to candleIndex
                        int proofTime = array.get(savedTimes, proofIndex)
                        if proofTime >= reverseMiddleTime
                            float proofClose = array.get(savedCloses, proofIndex)
                            bool atAnchor = wantUp ? proofClose - reverseFirst >= offset - 0.000000001 : reverseFirst - proofClose >= offset - 0.000000001
                            bool atMiddle = wantUp ? reverseMiddle - proofClose >= offset - 0.000000001 : proofClose - reverseMiddle >= offset - 0.000000001
                            if proofTime - proofPreviousTime == 300000
                                anchorBreak := anchorBreak or (atAnchor and previousAnchor)
                                middleBreak := middleBreak or (atMiddle and previousMiddle)
                            previousAnchor := atAnchor
                            previousMiddle := atMiddle
                            proofPreviousTime := proofTime
                    if middleBreak
                        reverseStage := 0
                    else if anchorBreak
                        trend := -trend
                        reverseStage := 0
                array.set(candleTrends, candleIndex, trend)
                result := trend
    // Только после состоявшегося закрепа перекрашиваем весь новый импульс от1.
    // До подтверждения участок сохраняет прежний цвет; от0 до1 он не меняется.
    if count > 0
        for paintIndex = 0 to count - 1
            int paintTrend = array.get(candleTrends, paintIndex)
            int paintTime = array.get(savedTimes, paintIndex)
            if paintTrend != 0
                if initialZeroBroken and paintTime > firstTime and paintTime <= zeroBreakColorTime
                    paintTrend := -initialTrend
                if paintTrend == 1
                    appendColoredCandle(greenPoints, paintTime, array.get(savedOpens, paintIndex), array.get(savedCloses, paintIndex))
                else
                    appendColoredCandle(redPoints, paintTime, array.get(savedOpens, paintIndex), array.get(savedCloses, paintIndex))
    result

recordMarketSwing(array<int> times, array<float> prices, array<bool> highs, array<int> days, int atTime, float price, bool isHigh, int dayKey, int historyLimit = 120) =>
    int count = array.size(times)
    bool sameDay = count > 0 ? array.get(days, count - 1) == dayKey : false
    bool sameType = sameDay ? array.get(highs, count - 1) == isHigh : false
    if sameType
        float oldPrice = array.get(prices, count - 1)
        if isHigh ? price >= oldPrice : price <= oldPrice
            array.set(times, count - 1, atTime)
            array.set(prices, count - 1, price)
    else
        array.push(times, atTime)
        array.push(prices, price)
        array.push(highs, isHigh)
        array.push(days, dayKey)
    while array.size(times) > historyLimit
        array.shift(times)
        array.shift(prices)
        array.shift(highs)
        array.shift(days)
    array.size(times)

swingClass(float price, float previousPrice, bool isHigh) =>
    na(previousPrice) ? (isHigh ? "H" : "L") : math.abs(price - previousPrice) <= 0.000000001 ? (isHigh ? "EH" : "EL") : isHigh ? (price > previousPrice ? "HH" : "LH") : (price > previousPrice ? "HL" : "LL")

// Полная структура: локальная остановка, свеча экстремума + четыре закрытые M5 справа.
// Тела, не тени; left=1 обозначает локальный поворот. Кандидаты не рисуются заранее.
float marketPivotHigh = ta.pivothigh(bodyHigh, 1, 4)
float marketPivotLow = ta.pivotlow(bodyLow, 1, 4)
bool structureSpanValid = m5Chart and not na(time[5]) and time - time[5] == 1500000
int visualPivotDay = rangeDayKey[4]
bool visualPivotAllowed = rangeWeekday[4] and time[4] >= zeroSearchStart[4]
// Открытая сделка может дойти до TP после полуночи; разметка относится к её дню.
if structureSpanValid and array.size(anchorHistoryDays) > 0
    for scopeIndex = array.size(anchorHistoryDays) - 1 to 0
        int candidateDay = array.get(anchorHistoryDays, scopeIndex)
        int candidateStart = array.get(anchorZeroTimes, scopeIndex)
        int candidateEnd = array.get(anchorTakeProfitTimes, scopeIndex)
        bool openCarry = candidateDay == positionPlanDay and (positionState == 2 or positionExitEvent != 0)
        bool scopedCarry = not na(candidateEnd) ? time[4] <= candidateEnd : openCarry
        if time[4] >= candidateStart and scopedCarry
            visualPivotDay := candidateDay
            visualPivotAllowed := true
            break
if barstate.isconfirmed and structureSpanValid and visualPivotAllowed
    bool hasHigh = not na(marketPivotHigh)
    bool hasLow = not na(marketPivotLow)
    // Если одна свеча одновременно high и low, порядок внутри неё неизвестен.
    // После существующей опоры допустима только противоположная; первую такую свечу пропускаем.
    if hasHigh and hasLow
        int count = array.size(marketSwingTimes)
        bool sameDay = count > 0 ? array.get(marketSwingDays, count - 1) == visualPivotDay : false
        if sameDay
            bool previousIsHigh = array.get(marketSwingHighs, count - 1)
            hasHigh := not previousIsHigh
            hasLow := previousIsHigh
        else
            hasHigh := false
            hasLow := false
    if hasHigh
        recordMarketSwing(marketSwingTimes, marketSwingPrices, marketSwingHighs, marketSwingDays, time[4], marketPivotHigh, true, visualPivotDay)
        recordMarketSwing(structureHistoryTimes, structureHistoryPrices, structureHistoryHighs, structureHistoryDays, time[4], marketPivotHigh, true, visualPivotDay, 10000)
    else if hasLow
        recordMarketSwing(marketSwingTimes, marketSwingPrices, marketSwingHighs, marketSwingDays, time[4], marketPivotLow, false, visualPivotDay)
        recordMarketSwing(structureHistoryTimes, structureHistoryPrices, structureHistoryHighs, structureHistoryDays, time[4], marketPivotLow, false, visualPivotDay, 10000)


// Архив обязательных 0 и 1: не зависит от бюджета временных меток.
if barstate.isconfirmed and m5Chart and firstSwingDisplayLocked and structureDay == rangeDayKey and not na(confirmedZeroPrice)
    int anchorCount = array.size(anchorHistoryDays)
    bool sameAnchorDay = anchorCount > 0 ? array.get(anchorHistoryDays, anchorCount - 1) == rangeDayKey : false
    if not sameAnchorDay
        array.push(anchorHistoryDays, rangeDayKey)
        array.push(anchorZeroTimes, confirmedZeroTime)
        array.push(anchorZeroPrices, confirmedZeroPrice)
        array.push(anchorOneTimes, firstExtremeTime)
        array.push(anchorOnePrices, structureFirstPrice)
        array.push(anchorZeroHighs, confirmedZeroPrice > structureFirstPrice)
        array.push(anchorTakeProfitTimes, na)
        if array.size(anchorHistoryDays) > 1000
            array.shift(anchorHistoryDays)
            array.shift(anchorZeroTimes)
            array.shift(anchorZeroPrices)
            array.shift(anchorOneTimes)
            array.shift(anchorOnePrices)
            array.shift(anchorZeroHighs)
            array.shift(anchorTakeProfitTimes)
    else
        // Расчётная опора V1 может переноситься после 1, но это уже не начало 0→1.
        // Видимая 0 сохраняет противоположный тип и строго предшествует точке 1.
        bool chronologicalZero = confirmedZeroTime < firstExtremeTime
        bool oppositeZero = structureSide == 1 ? confirmedZeroPrice < structureFirstPrice : confirmedZeroPrice > structureFirstPrice
        if chronologicalZero and oppositeZero
            array.set(anchorZeroTimes, anchorCount - 1, confirmedZeroTime)
            array.set(anchorZeroPrices, anchorCount - 1, confirmedZeroPrice)
            array.set(anchorZeroHighs, anchorCount - 1, structureSide == -1)
        array.set(anchorOneTimes, anchorCount - 1, firstExtremeTime)
        array.set(anchorOnePrices, anchorCount - 1, structureFirstPrice)

if barstate.isconfirmed and m5Chart and not na(v12Zero) and v12Day==rangeDayKey
    int v12Archive = array.indexof(anchorHistoryDays,rangeDayKey)
    if v12Archive>=0
        array.set(anchorZeroTimes,v12Archive,v12ZeroTime)
        array.set(anchorZeroPrices,v12Archive,v12Zero)
        array.set(anchorOneTimes,v12Archive,v12OneTime)
        array.set(anchorOnePrices,v12Archive,v12One)
        array.set(anchorZeroHighs,v12Archive,v12InitialSide==-1)

// SL не завершает разметку; TP первой либо второй сделки закрывает её окно.
if barstate.isconfirmed and positionExitEvent == 1
    int takeProfitIndex = array.indexof(anchorHistoryDays, positionPlanDay)
    if takeProfitIndex >= 0
        array.set(anchorTakeProfitTimes, takeProfitIndex, time)

// На графике только 0/1 с H/L и структура HH/HL/LH/LL. Внутренние номера движка скрыты.
if barstate.islast
    while array.size(marketStructureLabels) > 0
        label.delete(array.pop(marketStructureLabels))
    // Метки движка нужны внутренним опорам, но их номера больше не выводим.
    if array.size(firstPointLabels) > 0
        for numberIndex = 0 to array.size(firstPointLabels) - 1
            label.set_text(array.get(firstPointLabels, numberIndex), "")
    int viewportLeft = chart.left_visible_bar_time
    int viewportRight = chart.right_visible_bar_time
    int visibleSwingCount = 0
    if array.size(structureHistoryTimes) > 0
        for historyIndex = 0 to array.size(structureHistoryTimes) - 1
            int historyTime = array.get(structureHistoryTimes, historyIndex)
            int historyDay = array.get(structureHistoryDays, historyIndex)
            int historyScope = array.indexof(anchorHistoryDays, historyDay)
            int historyStart = historyScope >= 0 ? array.get(anchorZeroTimes, historyScope) : na
            int historyEnd = historyScope >= 0 ? array.get(anchorTakeProfitTimes, historyScope) : na
            if historyTime >= viewportLeft and historyTime <= viewportRight and not na(historyStart) and historyTime >= historyStart and (na(historyEnd) or historyTime <= historyEnd)
                visibleSwingCount := visibleSwingCount + 1
    int visibleSwingSeen = 0
    float previousHigh = na
    float previousLow = na
    int previousSwingDay = na
    int scopedStart = na
    int scopedEnd = na
    float scopedZeroPrice = na
    bool scopedZeroHigh = false
    if showSwingClasses and array.size(structureHistoryTimes) > 0
        for swingIndex = 0 to array.size(structureHistoryTimes) - 1
            int swingDay = array.get(structureHistoryDays, swingIndex)
            if na(previousSwingDay) or swingDay != previousSwingDay
                int scopeIndex = array.indexof(anchorHistoryDays, swingDay)
                scopedStart := scopeIndex >= 0 ? array.get(anchorZeroTimes, scopeIndex) : na
                scopedEnd := scopeIndex >= 0 ? array.get(anchorTakeProfitTimes, scopeIndex) : na
                scopedZeroPrice := scopeIndex >= 0 ? array.get(anchorZeroPrices, scopeIndex) : na
                scopedZeroHigh := scopeIndex >= 0 ? array.get(anchorZeroHighs, scopeIndex) : false
                previousHigh := scopedZeroHigh ? scopedZeroPrice : na
                previousLow := scopedZeroHigh ? na : scopedZeroPrice
                previousSwingDay := swingDay
            bool isHigh = array.get(structureHistoryHighs, swingIndex)
            float price = array.get(structureHistoryPrices, swingIndex)
            int swingTime = array.get(structureHistoryTimes, swingIndex)
            bool inScope = not na(scopedStart) and swingTime >= scopedStart and (na(scopedEnd) or swingTime <= scopedEnd)
            bool zeroMatchesSwing = swingTime == scopedStart and isHigh == scopedZeroHigh and math.abs(price - scopedZeroPrice) <= 0.000000001
            string classification = swingClass(price, zeroMatchesSwing ? na : isHigh ? previousHigh : previousLow, isHigh)
            string caption = classification == "H" ? "HH" : classification == "L" ? "LL" : classification
            bool structuralCaption = isHigh ? na(previousHigh) or price > previousHigh : na(previousLow) or price < previousLow
            bool inViewport = inScope and swingTime >= viewportLeft and swingTime <= viewportRight
            if inViewport
                visibleSwingSeen := visibleSwingSeen + 1
            if inViewport and visibleSwingSeen > math.max(0, visibleSwingCount - 116)
                if structuralCaption
                    label structureLabel = label.new(x=swingTime, y=price, text=caption, xloc=xloc.bar_time, yloc=yloc.price, style=isHigh ? label.style_label_down : label.style_label_up, color=color.new(color.white, 100), textcolor=color.black, size=size.small)
                    array.push(marketStructureLabels, structureLabel)
            if inScope and structuralCaption
                if isHigh
                    previousHigh := price
                else
                    previousLow := price

    if array.size(anchorHistoryDays) > 0
        for anchorIndex = 0 to array.size(anchorHistoryDays) - 1
            for anchorNumber = 0 to 1
                int anchorTime = anchorNumber == 0 ? array.get(anchorZeroTimes, anchorIndex) : array.get(anchorOneTimes, anchorIndex)
                float anchorPrice = anchorNumber == 0 ? array.get(anchorZeroPrices, anchorIndex) : array.get(anchorOnePrices, anchorIndex)
                bool anchorHigh = anchorNumber == 0 ? array.get(anchorZeroHighs, anchorIndex) : not array.get(anchorZeroHighs, anchorIndex)
                int anchorEnd = array.get(anchorTakeProfitTimes, anchorIndex)
                if anchorTime >= viewportLeft and anchorTime <= viewportRight and (na(anchorEnd) or anchorTime <= anchorEnd)
                    bool foundAnchor = false
                    if not foundAnchor and array.size(marketStructureLabels) > 0
                        for labelIndex = 0 to array.size(marketStructureLabels) - 1
                            label existingSwing = array.get(marketStructureLabels, labelIndex)
                            if label.get_x(existingSwing) == anchorTime and math.abs(label.get_y(existingSwing) - anchorPrice) <= 0.000000001
                                label.set_text(existingSwing, str.tostring(anchorNumber) + (anchorHigh ? " · H" : " · L"))
                                foundAnchor := true
                                break
                    if not foundAnchor
                        label anchorLabel = label.new(x=anchorTime, y=anchorPrice, text=str.tostring(anchorNumber) + (anchorHigh ? " · H" : " · L"), xloc=xloc.bar_time, yloc=yloc.price, style=anchorHigh ? label.style_label_down : label.style_label_up, color=color.new(color.white, 100), textcolor=color.black, size=size.small)
                        array.push(marketStructureLabels, anchorLabel)
                        while array.size(marketStructureLabels) > 120
                            label.delete(array.shift(marketStructureLabels))
    while array.size(marketStructureLabels) > 120
        label.delete(array.shift(marketStructureLabels))
