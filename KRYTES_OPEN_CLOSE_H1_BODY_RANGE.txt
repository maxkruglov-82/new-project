// Свеча экстремума — первая; считаем все закрытые M5 независимо от цвета.
nextRetreatCount(int previousCount, bool contiguous, float candleOpen, float candleClose, bool upward) =>
    contiguous ? previousCount + 1 : 1

// Закрепы за всеми сформированными точками: два close M5 подряд.
// Каждое закрытие минимум на offset за одним и тем же уровнем.
// За сформированной 1 тоже два; только первый выход за час — одно закрытие.
twoM5Break(float level, bool upward, float offset, bool adjacentSameDay) =>
    bool currentBeyond = upward ? close - level >= offset - 0.000000001 : level - close >= offset - 0.000000001
    bool previousBeyond = upward ? close[1] - level >= offset - 0.000000001 : level - close[1] >= offset - 0.000000001
    adjacentSameDay and currentBeyond and previousBeyond

// Точка 0 — экстремум собственного отката от противоположного экстремума.
// От 13:00 строго до 1; минимум пять свечей от начала отката включительно.
findRangeOrigin(array<float> savedOpens, array<float> savedCloses, array<int> savedTimes, int wantedSide, int fromTime, int beforeFirst) =>
    float foundPrice = na
    int foundTime = na
    int bestSpan = -1
    int count = array.size(savedTimes)
    if wantedSide == 1 and count > 1
        for anchorIndex = 0 to count - 2
            int anchorTime = array.get(savedTimes, anchorIndex)
            float anchorOpen = array.get(savedOpens, anchorIndex)
            float anchorClose = array.get(savedCloses, anchorIndex)
            float anchorPrice = wantedSide == 1 ? math.min(anchorOpen, anchorClose) : math.max(anchorOpen, anchorClose)
            bool localAnchor = anchorIndex == 0
            if anchorIndex > 0
                float previousOpen = array.get(savedOpens, anchorIndex - 1)
                float previousClose = array.get(savedCloses, anchorIndex - 1)
                localAnchor := wantedSide == 1 ? anchorPrice < math.min(previousOpen, previousClose) : anchorPrice > math.max(previousOpen, previousClose)
            float nextClose = array.get(savedCloses, anchorIndex + 1)
            bool startsRetreat = array.get(savedTimes, anchorIndex + 1) - anchorTime == 300000 and (wantedSide == 1 ? nextClose >= anchorPrice : nextClose <= anchorPrice)
            if anchorTime >= fromTime and anchorTime < beforeFirst and localAnchor and startsRetreat
                float retreatExtreme = anchorPrice
                int retreatTime = anchorTime
                int retreatSpan = 1
                bool following = true
                for rightIndex = anchorIndex + 1 to count - 1
                    if following
                        int rightTime = array.get(savedTimes, rightIndex)
                        if rightTime >= beforeFirst or rightTime - anchorTime != (rightIndex - anchorIndex) * 300000
                            following := false
                        else
                            float rightOpen = array.get(savedOpens, rightIndex)
                            float rightClose = array.get(savedCloses, rightIndex)
                            float rightExtreme = wantedSide == 1 ? math.max(rightOpen, rightClose) : math.min(rightOpen, rightClose)
                            bool reachesExtreme = wantedSide == 1 ? rightExtreme >= retreatExtreme : rightExtreme <= retreatExtreme
                            if reachesExtreme
                                retreatExtreme := rightExtreme
                                retreatTime := rightTime
                                retreatSpan := rightIndex - anchorIndex + 1
                            // Открытие пробойной свечи ещё может повторить экстремум.
                            if wantedSide == 1 ? rightClose < anchorPrice : rightClose > anchorPrice
                                following := false
                bool betterPrice = na(foundPrice) or (wantedSide == 1 ? retreatExtreme > foundPrice : retreatExtreme < foundPrice)
                if retreatExtreme > anchorPrice and retreatSpan >= 5 and (retreatSpan > bestSpan or (retreatSpan == bestSpan and betterPrice))
                    foundPrice := retreatExtreme
                    foundTime := retreatTime
                    bestSpan := retreatSpan
    // LOW 0: проверяем КАЖДУЮ свечу от 13:00, а не только локальные хаи.
    // Участок заканчивается перед первым обновлением максимума тела.
    // Начальная свеча входит в длину (минимум 5), но её low не кандидат.
    if wantedSide == -1 and count > 1
        for scanIndex = 0 to count - 2
            int scanTime = array.get(savedTimes, scanIndex)
            if scanTime >= fromTime and scanTime < beforeFirst
                float ceiling = math.max(array.get(savedOpens, scanIndex), array.get(savedCloses, scanIndex))
                float candidate = na
                int candidateTime = na
                int span = 1
                bool scanning = true
                for nextIndex = scanIndex + 1 to count - 1
                    if scanning
                        int nextTime = array.get(savedTimes, nextIndex)
                        float nextOpen = array.get(savedOpens, nextIndex)
                        float nextClose = array.get(savedCloses, nextIndex)
                        if nextTime >= beforeFirst or nextTime - scanTime != (nextIndex - scanIndex) * 300000 or math.max(nextOpen, nextClose) > ceiling
                            scanning := false
                        else
                            span := span + 1
                            float nextLow = math.min(nextOpen, nextClose)
                            if na(candidate) or nextLow <= candidate
                                candidate := nextLow
                                candidateTime := nextTime
                // Более поздний и одновременно более низкий low заменяет ранний.
                // При равном low выбираем последнее повторение, как в примере 21.10.
                bool betterLow = na(foundPrice) or candidate < foundPrice or (candidate == foundPrice and candidateTime > foundTime)
                if span >= 5 and not na(candidate) and betterLow
                    foundPrice := candidate
                    foundTime := candidateTime
    [foundPrice, foundTime]

// Внутренний формат: четыре точки тела и служебный разделитель.
// drawColoredRuns удаляет разделитель и разделяет объекты по разрывам времени.
appendColoredCandle(array<chart.point> points, int candleTime, float candleOpen, float candleClose) =>
    array.push(points, chart.point.from_time(candleTime - 60000, candleOpen))
    array.push(points, chart.point.from_time(candleTime, candleOpen))
    array.push(points, chart.point.from_time(candleTime, candleClose))
    array.push(points, chart.point.from_time(candleTime + 60000, candleClose))
    array.push(points, chart.point.from_time(candleTime + 120000, na))

// Отдельная ломаная на непрерывный участок одного цвета.
// Не полагаемся на na: терминал может соединять соседние конечные точки.
saveColoredRun(array<chart.point> run, array<polyline> drawings, array<polyline> history, color ink) =>
    while array.size(history) + array.size(drawings) >= 94
        if array.size(history) > 0
            polyline.delete(array.shift(history))
        else
            polyline.delete(array.shift(drawings))
    array.push(drawings, polyline.new(run, xloc=xloc.bar_time, line_color=ink, line_width=2))

drawColoredRuns(array<chart.point> points, array<polyline> drawings, array<polyline> history, color ink) =>
    array<chart.point> run = array.new<chart.point>()
    int previousCandle = na
    int count = array.size(points)
    if count >= 5
        for candleIndex = 0 to int(count / 5) - 1
            int base = candleIndex * 5
            chart.point firstPoint = array.get(points, base)
            int candleStart = firstPoint.time
            if not na(previousCandle) and candleStart - previousCandle != 300000
                saveColoredRun(run, drawings, history, ink)
                array.clear(run)
            for component = 0 to 3
                array.push(run, array.get(points, base + component))
            previousCandle := candleStart
        if array.size(run) > 0
            saveColoredRun(run, drawings, history, ink)
    array.size(drawings)

// Базовый тренд 0–1 сохраняется до закрепа за 0. После смены
// откат сохраняет цвет; новая опора 3 защищает подтверждённый участок 3–4.
rebuildColoredDay(array<chart.point> redPoints, array<chart.point> greenPoints, array<float> savedOpens, array<float> savedCloses, array<int> savedTimes, int zeroTime, float zeroPrice, float firstPrice, int originBreakTime, int fourthBreakTime, float thirdPrice, float fourthPrice, bool fifthReady, int fifthTime, float fifthPrice, int fifthSide, float offset, int endTime) =>
    array.clear(redPoints)
    array.clear(greenPoints)
    int initialSide = firstPrice > zeroPrice ? 1 : -1
    int trend = initialSide
    float floorPrice = math.min(zeroPrice, firstPrice)
    float ceilingPrice = math.max(zeroPrice, firstPrice)
    bool originApplied = false
    bool fourthApplied = false
    int result = 0
    int count = array.size(savedTimes)
    if count > 0
        for candleIndex = 0 to count - 1
            int candleTime = array.get(savedTimes, candleIndex)
            float candleOpen = array.get(savedOpens, candleIndex)
            float candleClose = array.get(savedCloses, candleIndex)
            if candleTime >= zeroTime and (na(endTime) or candleTime <= endTime)
                if not originApplied and not na(originBreakTime) and candleTime >= originBreakTime
                    originApplied := true
                    trend := -initialSide
                    floorPrice := trend == 1 ? firstPrice : math.min(candleOpen, candleClose)
                    ceilingPrice := trend == 1 ? math.max(candleOpen, candleClose) : firstPrice
                if originApplied
                    if not na(fourthBreakTime) and candleTime >= fourthBreakTime and not fourthApplied
                        fourthApplied := true
                        if fourthPrice > thirdPrice
                            floorPrice := thirdPrice
                            ceilingPrice := math.max(ceilingPrice, math.max(candleOpen, candleClose))
                        else
                            ceilingPrice := thirdPrice
                            floorPrice := math.min(floorPrice, math.min(candleOpen, candleClose))
                    bool previousAtFloor = false
                    bool previousAtCeiling = false
                    if candleIndex > 0
                        int previousTime = array.get(savedTimes, candleIndex - 1)
                        float previousClose = array.get(savedCloses, candleIndex - 1)
                        bool pairInRun = candleTime - previousTime == 300000 and previousTime >= zeroTime
                        previousAtFloor := pairInRun and floorPrice - previousClose >= offset - 0.000000001
                        previousAtCeiling := pairInRun and previousClose - ceilingPrice >= offset - 0.000000001
                    if trend == 1 and floorPrice - candleClose >= offset - 0.000000001 and previousAtFloor
                        trend := -1
                    else if trend == -1 and candleClose - ceilingPrice >= offset - 0.000000001 and previousAtCeiling
                        trend := 1
                    if trend == 1
                        ceilingPrice := math.max(ceilingPrice, math.max(candleOpen, candleClose))
                    else
                        floorPrice := math.min(floorPrice, math.min(candleOpen, candleClose))
                    if fifthReady and candleTime >= fifthTime
                        if fifthSide == 1
                            ceilingPrice := fifthPrice
                        else
                            floorPrice := fifthPrice
                if trend == 1
                    appendColoredCandle(greenPoints, candleTime, candleOpen, candleClose)
                else
                    appendColoredCandle(redPoints, candleTime, candleOpen, candleClose)
                result := trend
    result

// Подписи уровней как в v72: прозрачный фон, цвет уровня, плюс по результату.
updatePositionPrice(label id, int atTime, float price, string caption, color ink) =>
    label result = id
    if na(result)
        result := label.new(x=atTime, y=price, text=caption, xloc=xloc.bar_time, yloc=yloc.price, style=label.style_label_left, color=color.new(color.white, 100), textcolor=ink, size=size.small)
    else
        label.set_xy(result, atTime, price)
        label.set_text(result, caption)
    result

deletePositionPrice(array<label> labels, label info) =>
    int infoIndex = array.indexof(labels, info)
    if infoIndex >= 0
        array.remove(labels, infoIndex)
    label.delete(info)

// Отменённое ожидание удаляется целиком, исполненные сделки сохраняются.
removePendingPlan(array<box> boxes, array<label> labels, box riskBox, box rewardBox, label info, label stopInfo, label targetInfo) =>
    int riskIndex = array.indexof(boxes, riskBox)
    if riskIndex >= 0
        array.remove(boxes, riskIndex)
    int rewardIndex = array.indexof(boxes, rewardBox)
    if rewardIndex >= 0
        array.remove(boxes, rewardIndex)
    box.delete(riskBox)
    box.delete(rewardBox)
    deletePositionPrice(labels, info)
    deletePositionPrice(labels, stopInfo)
    deletePositionPrice(labels, targetInfo)

            positionTarget := positionSide == 1 ? math.max(positionTarget, currentTarget) : math.min(positionTarget, currentTarget)
            positionThreshold := (positionStop + positionTarget) / 2.0
            bool stopBeforeEntry = positionSide == 1 ? bodyLow <= positionStop : bodyHigh >= positionStop
            if bar_index > positionSetupBar and stopBeforeEntry
                positionState := 5
            else if bar_index > positionSetupBar
                float riskAtClose = positionSide == 1 ? close - positionStop : positionStop - close
                float rewardAtClose = positionSide == 1 ? positionTarget - close : close - positionTarget
                bool originContinuation = not positionIsZero and not positionIsSecond and not positionUsesFourth and not positionUsesFifth and not na(confirmedZeroPrice)
                bool touchedRRZone = positionSide == 1 ? (originContinuation ? low : bodyLow) <= positionThreshold : (originContinuation ? high : bodyHigh) >= positionThreshold
                bool againstInitialTrend = not na(confirmedZeroPrice) and (structureFirstPrice > confirmedZeroPrice ? positionSide == -1 : positionSide == 1)
                bool originConfirmed = not againstInitialTrend or zeroTrendReversed or (positionIsSecond and firstStopped and not rangeFriday)
                if dailyPositionEntries < 2 and (not positionIsSecond or firstStopped) and originConfirmed and riskAtClose > 0 and rewardAtClose > 0 and touchedRRZone
                    dailyPositionEntries := dailyPositionEntries + 1
                    positionEntry := close
                    positionRR := rewardAtClose / riskAtClose
                    positionEntryBar := bar_index
                    positionEntryEvent := true
                    colorStopTime := na
                    positionState := 2
    // Внутрисвечное движение ДО входа по close не является результатом сделки.
    if positionState == 2 and bar_index > positionEntryBar
        bool hitStop = positionSide == 1 ? bodyLow <= positionStop : bodyHigh >= positionStop
        bool hitTarget = positionSide == 1 ? high >= positionTarget : low <= positionTarget
        if hitStop or hitTarget
            colorStopTime := time
            positionExitEvent := hitStop ? -1 : 1
            positionState := hitStop ? 4 : 3
            // Фиксируем подписи предыдущей сделки до возможного создания второй.
            positionStopInfo := updatePositionPrice(positionStopInfo, time + 600000, positionStop, "Стоп: " + str.tostring(positionStop, "#.#####") + (positionState == 4 ? " +" : ""), color.red)
            positionTargetInfo := updatePositionPrice(positionTargetInfo, time + 600000, positionTarget, "Цель: " + str.tostring(positionTarget, "#.#####") + (positionState == 3 ? " +" : ""), color.teal)
            if positionState == 4 and dailyPositionEntries == 1 and positionPlanDay == rangeDayKey
                firstStopped := true
                firstStopBar := bar_index
                if rangeFriday
                    fridayFirstPrice := firstPositionSide == -1 ? bodyLow : bodyHigh
                    fridayFirstTime := time
                    fridayFirstBar := bar_index
                    fridayRetreatCount := 1
                    fridayFirstReady := false
                    fridayPullbackPrice := na
                    fridayPullbackTime := na
    // Пятница: отдельная новая структура в сторону первой сделки.
    bool fridayBreak = false
    if rangeFriday and firstPointWindow and positionPlanDay == rangeDayKey and firstStopped and dailyPositionEntries == 1 and positionState == 4 and not positionIsSecond and bar_index > firstStopBar
        float fridayCandidate = firstPositionSide == -1 ? bodyLow : bodyHigh
        bool extendsFriday = firstPositionSide == -1 ? close < fridayFirstPrice : close > fridayFirstPrice
        if not fridayFirstReady and extendsFriday
            fridayFirstPrice := fridayCandidate
            fridayFirstTime := time
            fridayFirstBar := bar_index
            fridayRetreatCount := 1
            fridayPullbackPrice := na
            fridayPullbackTime := na
        else
            float fridayOpposite = firstPositionSide == -1 ? bodyHigh : bodyLow
            if na(fridayPullbackPrice) or (firstPositionSide == -1 ? fridayOpposite > fridayPullbackPrice : fridayOpposite < fridayPullbackPrice)
                fridayPullbackPrice := fridayOpposite
                fridayPullbackTime := time
            if not fridayFirstReady
                fridayRetreatCount := nextRetreatCount(fridayRetreatCount, time - time[1] == 300000, open, close, firstPositionSide == -1)
                if fridayRetreatCount >= minimumRetreatBars
                    fridayFirstReady := true
            else
                fridayBreak := twoM5Break(fridayFirstPrice, firstPositionSide == 1, firstPointOffset, pointPairM5)
    // Остальные дни: прежняя противоположная сделка после закрепа за стопом.
    // Стоп за любым сформированным уровнем подтверждается двумя close.
    bool reverseBreak = twoM5Break(firstPositionStop, firstPositionSide == -1, firstPointOffset, pointPairM5)
    if firstPointWindow and positionPlanDay == rangeDayKey and firstStopped and dailyPositionEntries == 1 and positionState == 4 and not positionIsSecond and ((not rangeFriday and reverseBreak) or fridayBreak)
        // Завершаем рисунок первой позиции до создания нового инструмента.
        if not na(positionInfo)
            label.set_xy(positionInfo, time + 600000, positionEntry)
            label.set_text(positionInfo, "Вход: " + str.tostring(positionEntry, "#.#####") + " · RR 1:" + str.tostring(positionRR, "#.##"))
            box.set_right(positionRiskBox, time + 600000)
            box.set_right(positionRewardBox, time + 600000)
        positionSide := rangeFriday ? firstPositionSide : -firstPositionSide
        positionIsSecond := true
        positionUsesFifth := false
        positionIsZero := false
        if firstPositionUsesFifth or firstPositionIsZero
            fifthLabel := na
        positionAnchorTime := rangeFriday ? fridayPullbackTime : (firstPositionIsZero or firstPositionUsesFifth) ? zeroReversalTime : firstPositionUsesFourth ? fourthColorTime : thirdColorTime
        positionStop := rangeFriday ? fridayPullbackPrice : (firstPositionIsZero or firstPositionUsesFifth) ? zeroReversalStop : firstPositionUsesFourth ? confirmedFourthPrice : confirmedThirdPrice
        confirmedFifthPrice := positionSide == 1 ? bodyHigh : bodyLow
        positionTarget := confirmedFifthPrice
        positionThreshold := (positionStop + positionTarget) / 2.0
        positionState := 1
        positionSetupBar := bar_index
        positionEntryBar := na
        positionEntry := na
        positionRR := na
        positionRiskBox := na
        positionRewardBox := na
        positionInfo := na
        positionStopInfo := na
        positionTargetInfo := na
        newPositionPlan := true
        fifthAnchorOpen := open
        fifthExtremeBar := bar_index
        fifthRetreatBars := 1
        fifthLocked := false
        fifthColorTime := time
        if not firstPositionIsZero and not firstPositionUsesFifth
            fifthLabel := label.new(x=time, y=confirmedFifthPrice, text=firstPositionUsesFourth ? (originRelocated ? "4" : "5") : (originRelocated ? "3" : "4"), xloc=xloc.bar_time, yloc=yloc.price, style=positionSide == 1 ? label.style_label_down : label.style_label_up, color=color.new(color.red, 100), textcolor=color.red, size=size.large)
            array.push(firstPointLabels, fifthLabel)
            if array.size(firstPointLabels) > 360
                label.delete(array.shift(firstPointLabels))
    if positionState == 5 and positionIsZero and dailyPositionEntries == 0
        zeroScenarioCancelled := true
    if positionState == 5 and not na(positionInfo)
        removePendingPlan(positionBoxes, positionLabels, positionRiskBox, positionRewardBox, positionInfo, positionStopInfo, positionTargetInfo)
        positionRiskBox := na
        positionRewardBox := na
        positionInfo := na
        positionStopInfo := na
        positionTargetInfo := na
    // После SL сторона нового сценария могла измениться на этой же M5.
    pendingAgainstTrend := not na(confirmedZeroPrice) and (structureFirstPrice > confirmedZeroPrice ? positionSide == -1 : positionSide == 1)
    pendingDirectionAllowed := not pendingAgainstTrend or zeroTrendReversed or (positionIsSecond and firstStopped and not rangeFriday)
    if newPositionPlan and pendingDirectionAllowed
        positionRiskBox := box.new(left=(na(positionAnchorTime) ? time : positionAnchorTime) - 600000, top=math.max(positionStop, positionThreshold), right=time + 600000, bottom=math.min(positionStop, positionThreshold), xloc=xloc.bar_time, border_color=color.red, bgcolor=color.new(color.red, 85))
        positionRewardBox := box.new(left=(na(positionAnchorTime) ? time : positionAnchorTime) - 600000, top=math.max(positionTarget, positionThreshold), right=time + 600000, bottom=math.min(positionTarget, positionThreshold), xloc=xloc.bar_time, border_color=color.teal, bgcolor=color.new(color.teal, 85))
        positionInfo := updatePositionPrice(na, time + 600000, positionThreshold, "Граница RR1: " + str.tostring(positionThreshold, "#.#####"), color.blue)
        positionStopInfo := updatePositionPrice(na, time + 600000, positionStop, "Стоп: " + str.tostring(positionStop, "#.#####"), color.red)
        positionTargetInfo := updatePositionPrice(na, time + 600000, positionTarget, "Цель: " + str.tostring(positionTarget, "#.#####"), color.teal)
        array.push(positionBoxes, positionRiskBox)
        array.push(positionBoxes, positionRewardBox)
        array.push(positionLabels, positionInfo)
        array.push(positionLabels, positionStopInfo)
        array.push(positionLabels, positionTargetInfo)
        // 40 инструментов по 3 подписи + 360 структурных меток: максимум 480.
        while array.size(positionBoxes) > 80
            box.delete(array.shift(positionBoxes))
        while array.size(positionLabels) > 120
            label.delete(array.shift(positionLabels))
    if not na(positionInfo) and (positionState == 1 or positionState == 2 or positionEntryEvent or positionState != positionState[1])
        float drawnEntry = na(positionEntry) ? positionThreshold : positionEntry
        box.set_top(positionRiskBox, math.max(positionStop, drawnEntry))
        box.set_bottom(positionRiskBox, math.min(positionStop, drawnEntry))
        box.set_top(positionRewardBox, math.max(positionTarget, drawnEntry))
        box.set_bottom(positionRewardBox, math.min(positionTarget, drawnEntry))
        box.set_right(positionRiskBox, time + 600000)
        box.set_right(positionRewardBox, time + 600000)
        float entryCaptionRisk = math.abs(drawnEntry - positionStop)
        string entryCaptionRR = entryCaptionRisk > 0 ? " · RR 1:" + str.tostring(math.abs(positionTarget - drawnEntry) / entryCaptionRisk, "#.##") : ""
        positionInfo := updatePositionPrice(positionInfo, time + 600000, drawnEntry, (na(positionEntry) ? "Граница RR1: " : "Вход: ") + str.tostring(drawnEntry, "#.#####") + entryCaptionRR + (positionState == 6 ? " · TP/SL?" : ""), color.blue)
        positionStopInfo := updatePositionPrice(positionStopInfo, time + 600000, positionStop, "Стоп: " + str.tostring(positionStop, "#.#####") + (positionState == 4 ? " +" : ""), color.red)
        positionTargetInfo := updatePositionPrice(positionTargetInfo, time + 600000, positionTarget, "Цель: " + str.tostring(positionTarget, "#.#####") + (positionState == 3 ? " +" : ""), color.teal)
        float displayedRisk = math.abs(drawnEntry - positionStop)
        string rrCaption = displayedRisk > 0 ? "R:R 1:" + str.tostring(math.abs(positionTarget - drawnEntry) / displayedRisk, "#.###") : ""
        box.set_text(positionRewardBox, rrCaption)
        box.set_text_color(positionRewardBox, color.white)
        box.set_text_size(positionRewardBox, size.small)
