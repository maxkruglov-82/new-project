// 0 — крайняя цена тела внутри H1, по времени от 14:00 строго до 1.
// При равной цене сохраняем первое достижение экстремума.
findRangeOrigin(array<float> savedOpens, array<float> savedCloses, array<int> savedTimes, int wantedSide, float upperBound, float lowerBound, int fromTime, int beforeFirst) =>
    float foundPrice = na
    int foundTime = na
    int count = array.size(savedTimes)
    if count > 0
        for scanIndex = 0 to count - 1
            int scanTime = array.get(savedTimes, scanIndex)
            if scanTime >= fromTime and scanTime < beforeFirst
                float scanOpen = array.get(savedOpens, scanIndex)
                float scanClose = array.get(savedCloses, scanIndex)
                float candidate = wantedSide == 1 ? math.max(scanOpen, scanClose) : math.min(scanOpen, scanClose)
                if candidate >= lowerBound and candidate <= upperBound
                    if na(foundPrice) or (wantedSide == 1 ? candidate > foundPrice : candidate < foundPrice)
                        foundPrice := candidate
                        foundTime := scanTime
    [foundPrice, foundTime]

// Окраска уже сформированных участков. Исходные свечи/теневые экстремумы не меняются.
paintTrendSegment(array<line> segmentLines, array<line> historyLines, array<float> candleOpens, array<float> candleCloses, array<int> candleTimes, int fromTime, float fromPrice, int toTime, float toPrice, color risingColor, color fallingColor, bool includeStart) =>
    while array.size(segmentLines) > 0
        line previousLine = array.pop(segmentLines)
        int retainedIndex = array.indexof(historyLines, previousLine)
        if retainedIndex >= 0
            array.remove(historyLines, retainedIndex)
            line.delete(previousLine)
    int paintedBars = 0
    if not na(fromTime) and not na(toTime) and not na(fromPrice) and not na(toPrice) and toTime >= fromTime and toPrice != fromPrice
        color trendInk = toPrice > fromPrice ? risingColor : fallingColor
        int candleCount = array.size(candleTimes)
        if candleCount > 0
            for candleIndex = 0 to candleCount - 1
                int candleTime = array.get(candleTimes, candleIndex)
                if (includeStart ? candleTime >= fromTime : candleTime > fromTime) and candleTime <= toTime
                    float candleOpen = array.get(candleOpens, candleIndex)
                    float candleClose = array.get(candleCloses, candleIndex)
                    // Запас до лимита 500 линий, без неограниченного накопления истории.
                    while array.size(historyLines) + 3 > 450
                        line.delete(array.shift(historyLines))
                    line bodyStroke = line.new(x1=candleTime, y1=math.min(candleOpen, candleClose), x2=candleTime, y2=math.max(candleOpen, candleClose), xloc=xloc.bar_time, color=trendInk, width=2)
                    line openStroke = line.new(x1=candleTime - 60000, y1=candleOpen, x2=candleTime, y2=candleOpen, xloc=xloc.bar_time, color=trendInk, width=2)
                    line closeStroke = line.new(x1=candleTime, y1=candleClose, x2=candleTime + 60000, y2=candleClose, xloc=xloc.bar_time, color=trendInk, width=2)
                    array.push(historyLines, bodyStroke)
                    array.push(historyLines, openStroke)
                    array.push(historyLines, closeStroke)
                    array.push(segmentLines, bodyStroke)
                    array.push(segmentLines, openStroke)
                    array.push(segmentLines, closeStroke)
                    paintedBars := paintedBars + 1
    paintedBars

// Отменённое ожидание удаляется целиком, исполненные сделки сохраняются.
removePendingPlan(array<box> boxes, array<label> labels, box riskBox, box rewardBox, label info) =>
    int riskIndex = array.indexof(boxes, riskBox)
    if riskIndex >= 0
        array.remove(boxes, riskIndex)
    int rewardIndex = array.indexof(boxes, rewardBox)
    if rewardIndex >= 0
        array.remove(boxes, rewardIndex)
    int infoIndex = array.indexof(labels, info)
    if infoIndex >= 0
        array.remove(labels, infoIndex)
    box.delete(riskBox)
    box.delete(rewardBox)
    label.delete(info)
