indicator("SOL — часовой диапазон, SHORT и LONG 1:1 v10", overlay=true, max_boxes_count=500, max_labels_count=500, max_lines_count=500)

// График: SOL/USDT, MEXC Futures, 5 минут.
// Время: Красноярск UTC+7. Никаких ордеров — только разметка и alertcondition.
// На истории внутрибарный порядок неизвестен: свеча с новым экстремумом
// не используется одновременно как доказательство последующего ретеста.
string tz = "Asia/Krasnoyarsk"
bool showFibo = input.bool(true, "Показывать Фибоначчи")
bool showLevels = input.bool(true, "Показывать линии входа, стопа и цели")
bool showMarks = input.bool(true, "Показывать SHORT и LONG")
bool showPosition = input.bool(true, "Показывать зоны позиции")
int positionBars = input.int(24, "Ширина позиции (свечей M5)", minval=2, maxval=288)
int historyDays = input.int(60, "История: опорных дней", minval=1, maxval=60)
int historyUntil = input.time(0, "История до (0 — правый край экрана)")

// Восемь линий на сетап. Храним до 60 полных разметок в выбранном окне;
// при повторных попытках в один день самые старые группы удаляем целиком.
// Расчёт стратегии идёт по всем свечам; окно ограничивает только рисунки.
int historyAnchor = historyUntil != 0 ? historyUntil : nz(chart.right_visible_bar_time, last_bar_time)
// Видимый бар 14:55 заканчивается в 15:00; открытый бар не считаем закрытым.
int anchorMoment = historyUntil != 0 ? historyUntil : math.min(historyAnchor + 300000, timenow)
int anchorDate = dayofmonth(anchorMoment, tz) - (hour(anchorMoment, tz) < 15 ? 1 : 0)
int historyEnd = timestamp(tz, year(anchorMoment, tz), month(anchorMoment, tz), anchorDate, 15, 0)
int historyStart = historyEnd - (historyDays - 1) * 86400000

// Отдельные горизонтальные объекты не соединяют уровни разных дней.
// Не полагаемся на поддержку plot.style_linebr рендерером терминала.
updateLevel(line id, int leftTime, int rightTime, float price, color ink, int thickness) =>
    line result = id
    if na(result)
        result := line.new(leftTime, price, rightTime, price, xloc=xloc.bar_time, color=ink, width=thickness)
    else
        line.set_xy1(result, leftTime, price)
        line.set_xy2(result, rightTime, price)
    result

updateZone(box id, int leftTime, int rightTime, float firstPrice, float secondPrice, color ink, string caption) =>
    box result = id
    if na(result)
        result := box.new(left=leftTime, top=math.max(firstPrice, secondPrice), right=rightTime, bottom=math.min(firstPrice, secondPrice), xloc=xloc.bar_time, border_color=ink, bgcolor=color.new(ink, 85), text=caption, text_color=color.white, text_size=size.small)
    else
        box.set_left(result, leftTime)
        box.set_right(result, rightTime)
        box.set_top(result, math.max(firstPrice, secondPrice))
        box.set_bottom(result, math.min(firstPrice, secondPrice))
    result

updatePriceLabel(label id, int atTime, float price, string caption, color ink) =>
    label result = id
    if na(result)
        result := label.new(atTime, price, caption, xloc=xloc.bar_time, yloc=yloc.price, style=label.style_label_left, color=color.new(color.white, 100), textcolor=ink, size=size.small)
    else
        label.set_xy(result, atTime, price)
        label.set_text(result, caption)
    result

if barstate.isfirst and timeframe.period != "5"
    runtime.error("Выберите таймфрейм 5 минут. Часовой диапазон индикатор собирает автоматически.")

// Состояния: 0 нет диапазона; 1 ждём выход; 2 первый фрактал;
// 3 второй фрактал; 4 диапазон M5; 5 ретест; 6 вход/исход; 7 отмена.
// outcome: 0 ждём исход; 1 цель (блокировка H1); -1 стоп (новая попытка);
// 2 обе границы сделки на одной исторической свече (порядок неизвестен).
// varip сохраняет состояние между обновлениями открытой свечи.
varip int stage = 0
// -1 SHORT, +1 LONG, 0 направление пока не определено.
varip int direction = 0
varip bool drawReference = false
varip float h1High = na
varip float h1Low = na
varip int validUntil = na
varip int breakBar = na
varip float point1 = na
varip int point1Bar = na
varip int point1Time = na
varip float point2 = na
varip float point3 = na
varip float entry = na
varip float stop = na
varip int lastExtremeBar = na
varip int signalBar = na
varip float previousQuote = na
varip int lastPivotBar = na
varip int outcome = 0
varip int outcomeBar = na
varip int candidateEpoch = 0
varip string retiredStatus = ""
var int drawingEpoch = 0
// Очереди полных групп: 8 линий, 3 прямоугольника, 4 метки на кандидат.
var array<line> archivedLines = array.new_line()
var array<box> archivedBoxes = array.new_box()
var array<label> archivedLabels = array.new_label()

var box h1Box = na
var box m5Box = na
var line fibMinus = na
var line fibZero = na
var line fibHalf = na
var line fibOne = na
var line fibStop = na
var line entryLine = na
var line stopLine = na
var line targetLine = na
varip int drawingRight = na
varip int positionLeft = na
var box rewardZone = na
var box riskZone = na
var label entryPriceLabel = na
var label stopPriceLabel = na
var label targetPriceLabel = na
var label signalMark = na

// Сбор двенадцати закрытых M5-свечей 14:00–15:00.
// Неполный час не используется: нельзя подменять его частичным диапазоном.
var float collectedOpen = na
var float collectedHigh = na
var float collectedLow = na
var int collectedCount = 0
var int collectedDay = na
var int collectedTime = na
var int publishedDay = na

int yy = year(time, tz)
int mm = month(time, tz)
int dd = dayofmonth(time, tz)
int hh = hour(time, tz)
int mi = minute(time, tz)
int dayKey = yy * 10000 + mm * 100 + dd
bool shortEvent = false
bool longEvent = false
bool eventDraw = false
float eventEntry = na
float eventStop = na
float eventTarget = na

// Пока вход активен, противоположный выход H1 не создаёт вторую сделку.
// Исторический бар входа не доказывает последующий TP/SL: порядок неизвестен.
if stage == 6 and outcome == 0 and (barstate.isrealtime or bar_index > signalBar)
    // На следующих M5 вся тень уже относится к периоду после входа.
    // Даже без котировки самого касания high/low доказывает достигнутый
    // уровень. На свече входа старую тень не приписываем новой позиции.
    float reachedHigh = bar_index > signalBar ? high : close
    float reachedLow = bar_index > signalBar ? low : close
    bool stopHit = direction == -1 ? reachedHigh >= stop : reachedLow <= stop
    bool targetHit = direction == -1 ? reachedLow <= point3 : reachedHigh >= point3
    if stopHit or targetHit
        outcome := stopHit and targetHit ? 2 : targetHit ? 1 : -1
        outcomeBar := bar_index

// После стопа начинаем новую попытку со следующей M5. Не используем
// её предшественника задним числом как уже известный фрактал нового плана.
bool retry = stage == 6 and outcome == -1 and bar_index > outcomeBar and time < validUntil
if retry
    candidateEpoch := candidateEpoch + 1
    retiredStatus := ""
    stage := 1
    direction := 0
    outcome := 0
    outcomeBar := na
    breakBar := na
    point1 := na
    point1Bar := na
    point1Time := na
    point2 := na
    point3 := na
    entry := na
    stop := na
    signalBar := na
    lastExtremeBar := na
    drawingRight := na
    positionLeft := na

// Старый ожидающий сетап прекращается при новой опорной H1.
if stage >= 1 and stage <= 5
    if time >= validUntil
        stage := 0
    else
        bool below = barstate.isrealtime ? close < h1Low : low < h1Low
        bool above = barstate.isrealtime ? close > h1High : high > h1High
        if stage == 1
            // У исторической свечи с обоими пробоями нет известного порядка.
            if below and above
                stage := 7
            else if below or above
                direction := below ? -1 : 1
                stage := 2
                breakBar := bar_index

        // До входа противоположная граница меняет приоритет под той же H1.
        // Новый P1 — первый подтверждённый свинг на ноге нового пробоя.
        if stage >= 2 and stage <= 5 and ((direction == -1 and above) or (direction == 1 and below))
            candidateEpoch := candidateEpoch + 1
            retiredStatus := "Отмена · Без входа"
            direction := direction == -1 ? 1 : -1
            stage := below and above ? 7 : 2
            breakBar := bar_index
            point1 := na
            point1Bar := na
            point1Time := na
            point2 := na
            point3 := na
            entry := na
            stop := na
            signalBar := na
            lastExtremeBar := na
            drawingRight := na
            positionLeft := na

        // Вторую точку замораживаем в момент обновления первой.
        if stage == 4 and ((direction == -1 and low < point1) or (direction == 1 and high > point1))
            stop := point1 + 1.25 * (point2 - point1)
            // Для LONG стоп может лежать ниже low H1 (пример пользователя).
            // Отменяет LONG фактический пробой low H1 ценой, а не сам стоп.
            // Прежний отдельно согласованный фильтр SHORT сохраняется.
            if direction == -1 and stop > h1High
                stage := 7
            else
                point3 := direction == -1 ? low : high
                entry := (stop + point3) / 2.0
                lastExtremeBar := bar_index
                positionLeft := time
                stage := 5

        if stage == 5
            if (direction == -1 and low < point3) or (direction == 1 and high > point3)
                point3 := direction == -1 ? low : high
                entry := (stop + point3) / 2.0
                lastExtremeBar := bar_index

            // Текущая цена позволяет отличить ретест от high/low,
            // возникшего раньше нового экстремума открытой свечи.
            bool liveTouch = barstate.isrealtime and not na(previousQuote) and (direction == -1 ? previousQuote < entry and close >= entry : previousQuote > entry and close <= entry)
            // Без тиков на истории пропускаем неоднозначный порядок low/high.
            bool historyTouch = not barstate.isrealtime and barstate.isconfirmed and bar_index > lastExtremeBar and (direction == -1 ? open <= entry and high >= entry : open >= entry and low <= entry)
            if liveTouch or historyTouch
                stage := 6
                outcome := 0
                outcomeBar := na
                signalBar := bar_index
                shortEvent := direction == -1
                longEvent := direction == 1
                eventDraw := drawReference
                eventEntry := entry
                eventStop := stop
                eventTarget := point3

// Строгие трёхсвечные фракталы. Центральная свеча известна только после
// закрытия правой. Равные экстремумы не образуют новый фрактал.
bool lowerPivot = barstate.isconfirmed and bar_index >= 2 and low[1] < low[2] and low[1] < low
bool upperPivot = barstate.isconfirmed and bar_index >= 2 and high[1] > high[2] and high[1] > high

// Обновление фракталов происходит ПОСЛЕ обработки событий текущей свечи.
// Новый подтверждённый уровень не используется задним числом внутри неё.
if barstate.isconfirmed and (na(lastPivotBar) or lastPivotBar != bar_index)
    lastPivotBar := bar_index
    bool firstPivot = direction == -1 ? lowerPivot and low[1] < h1Low : upperPivot and high[1] > h1High
    bool secondPivot = direction == -1 ? upperPivot and high[1] > point1 : lowerPivot and low[1] < point1
    if stage == 2 and firstPivot and bar_index - 1 >= breakBar
        point1 := direction == -1 ? low[1] : high[1]
        point1Bar := bar_index - 1
        point1Time := time[1]
        stage := 3
    else if stage == 3
        // Пока нет второй точки, первая следует за более глубоким
        // подтверждённым свингом (для LONG — за более высоким).
        if firstPivot and ((direction == -1 and low[1] < point1) or (direction == 1 and high[1] > point1))
            point1 := direction == -1 ? low[1] : high[1]
            point1Bar := bar_index - 1
            point1Time := time[1]
        else if secondPivot and bar_index - 1 > point1Bar
            if bar_index - 1 == point1Bar + 1
                // Соседние центральные свечи не образуют диапазон 1–2.
                // Отбрасываем обе точки и ищем новый первый свинг.
                stage := 2
                breakBar := bar_index
                point1 := na
                point1Bar := na
                point1Time := na
                point2 := na
            else
                point2 := direction == -1 ? high[1] : low[1]
                stage := 4
    else if stage == 4 and ((direction == -1 and upperPivot and high[1] > point2) or (direction == 1 and lowerPivot and low[1] < point2))
        point2 := direction == -1 ? high[1] : low[1]

// Смена кандидата не двигает старые объекты. Обычный var откатывается
// на каждом тике; сравнение с varip epoch повторяет этот шаг после отката.
if drawingEpoch != candidateEpoch
    if not na(m5Box)
        if retiredStatus != "" and not na(entryPriceLabel)
            label.set_text(entryPriceLabel, retiredStatus + " · Вход: " + str.tostring(label.get_y(entryPriceLabel), "#.##"))
        array.push(archivedLines, fibMinus)
        array.push(archivedLines, fibZero)
        array.push(archivedLines, fibHalf)
        array.push(archivedLines, fibOne)
        array.push(archivedLines, fibStop)
        array.push(archivedLines, entryLine)
        array.push(archivedLines, stopLine)
        array.push(archivedLines, targetLine)
        array.push(archivedBoxes, m5Box)
        array.push(archivedBoxes, rewardZone)
        array.push(archivedBoxes, riskZone)
        array.push(archivedLabels, entryPriceLabel)
        array.push(archivedLabels, stopPriceLabel)
        array.push(archivedLabels, targetPriceLabel)
        array.push(archivedLabels, signalMark)
    m5Box := na
    fibMinus := na
    fibZero := na
    fibHalf := na
    fibOne := na
    fibStop := na
    entryLine := na
    stopLine := na
    targetLine := na
    rewardZone := na
    riskZone := na
    entryPriceLabel := na
    stopPriceLabel := na
    targetPriceLabel := na
    signalMark := na
    drawingEpoch := candidateEpoch

// Метка входит в ту же группу, что Фибо и позиция: при лимите истории
// удаляется вместе с ней, не оставляя одиночного SHORT/LONG без разметки.
if drawReference and showMarks and stage == 6 and not na(signalBar) and na(signalMark)
    signalMark := label.new(time, direction == -1 ? high : low, direction == -1 ? "SHORT" : "LONG", xloc=xloc.bar_time, yloc=direction == -1 ? yloc.abovebar : yloc.belowbar, style=direction == -1 ? label.style_label_down : label.style_label_up, color=direction == -1 ? color.red : color.green, textcolor=color.white, size=size.small)

// Рабочий диапазон M5. Его правый край сопровождает расчёт до сигнала/отмены.
if drawReference and stage >= 4 and stage <= 6 and not na(point1) and not na(point2)
    if na(m5Box)
        // Освобождаем место только перед новым видимым рисунком.
        // Если выбранное окно уже закончилось, все 60 архивных групп остаются.
        if array.size(archivedLines) >= 60 * 8
            for i = 0 to 7
                line.delete(array.shift(archivedLines))
            for i = 0 to 2
                box.delete(array.shift(archivedBoxes))
            for i = 0 to 3
                label.delete(array.shift(archivedLabels))
        m5Box := box.new(left=point1Time, top=math.max(point1, point2), right=time_close, bottom=math.min(point1, point2), xloc=xloc.bar_time, border_color=color.gray, border_style=line.style_dotted, bgcolor=na)
    if stage == 4
        box.set_top(m5Box, math.max(point1, point2))
        box.set_bottom(m5Box, math.min(point1, point2))
    if stage <= 5 or signalBar == bar_index
        box.set_right(m5Box, time_close)

bool baseVisible = showFibo and stage >= 5 and stage <= 6 and not na(point2)
bool planVisible = (stage == 5 or stage == 6) and not na(entry)
float span = point2 - point1
if stage >= 4 and stage <= 6 and (stage <= 5 or signalBar == bar_index)
    drawingRight := time_close
// Ширина — только размер рисунка, не срок действия сценария.
int positionRight = not na(positionLeft) ? math.max(positionLeft + positionBars * 300000, drawingRight) : na
if drawReference and showPosition and planVisible and not na(positionLeft)
    rewardZone := updateZone(rewardZone, positionLeft, positionRight, entry, point3, color.teal, "R:R 1:1")
    riskZone := updateZone(riskZone, positionLeft, positionRight, entry, stop, color.red, "")
    entryPriceLabel := updatePriceLabel(entryPriceLabel, positionRight, entry, "Вход: " + str.tostring(entry, "#.##"), color.blue)
    // Плюс отмечает только подтверждённый исход после входа. Неизвестный
    // порядок TP/SL и незавершённый план не получают отметку результата.
    stopPriceLabel := updatePriceLabel(stopPriceLabel, positionRight, stop, "Стоп: " + str.tostring(stop, "#.##") + (stage == 6 and outcome == -1 ? " +" : ""), color.red)
    targetPriceLabel := updatePriceLabel(targetPriceLabel, positionRight, point3, "Цель: " + str.tostring(point3, "#.##") + (stage == 6 and outcome == 1 ? " +" : ""), color.teal)
else if drawReference and (stage == 7 or stage == 0) and not na(entryPriceLabel)
    // Сохраняем неисполненный сетап для разбора, не приписывая ему вход.
    string status = stage == 7 ? "Отмена · Без входа" : "Истёк · Без входа"
    label.set_text(entryPriceLabel, status + " · Вход: " + str.tostring(entry, "#.##"))

if drawReference and stage >= 4 and stage <= 6
    if baseVisible
        fibMinus := updateLevel(fibMinus, point1Time, drawingRight, point1 - 0.25 * span, color.red, 1)
        fibZero := updateLevel(fibZero, point1Time, drawingRight, point1, color.gray, 1)
        fibHalf := updateLevel(fibHalf, point1Time, drawingRight, point1 + 0.5 * span, color.gray, 1)
        fibOne := updateLevel(fibOne, point1Time, drawingRight, point2, color.gray, 1)
        fibStop := updateLevel(fibStop, point1Time, drawingRight, point1 + 1.25 * span, color.red, 1)
    if showLevels and planVisible
        int levelRight = showPosition ? positionRight : drawingRight
        entryLine := updateLevel(entryLine, point1Time, levelRight, entry, color.blue, 2)
        stopLine := updateLevel(stopLine, point1Time, levelRight, stop, color.red, 2)
        targetLine := updateLevel(targetLine, point1Time, levelRight, point3, color.teal, 2)

// Новый диапазон публикуется в конце последней закрытой M5-свечи часа.
// До этого нельзя использовать high/low часовой свечи для сигналов.
if barstate.isconfirmed and hh == 14
    if mi == 0
        collectedOpen := open
        collectedHigh := high
        collectedLow := low
        collectedCount := 1
        collectedDay := dayKey
        collectedTime := time
    else if collectedDay == dayKey and time - collectedTime == 300000
        collectedHigh := math.max(collectedHigh, high)
        collectedLow := math.min(collectedLow, low)
        collectedCount := collectedCount + 1
        collectedTime := time
    else
        collectedCount := 0

    if mi == 55 and collectedCount == 12 and (na(publishedDay) or publishedDay != dayKey)
        // Старый рисунок остаётся на истории. Незавершённый план помечаем,
        // а ссылки переключаем на новый день, не удаляя прежние объекты.
        if drawReference and stage >= 1 and stage <= 5 and not na(entryPriceLabel)
            label.set_text(entryPriceLabel, "Без входа · Вход: " + str.tostring(entry, "#.##"))
        publishedDay := dayKey
        h1High := collectedHigh
        h1Low := collectedLow
        validUntil := timestamp(tz, yy, mm, dd + 1, 15, 0)
        stage := 1
        direction := 0
        int referenceClose = timestamp(tz, yy, mm, dd, 15, 0)
        drawReference := referenceClose >= historyStart and referenceClose <= historyEnd
        breakBar := na
        point1 := na
        point1Bar := na
        point1Time := na
        point2 := na
        point3 := na
        entry := na
        stop := na
        lastExtremeBar := na
        signalBar := na
        drawingRight := na
        positionLeft := na
        // Цены события хранятся отдельно: даже на границе смены H1
        // оповещение должно содержать уровни именно сработавшего плана.
        outcome := 0
        outcomeBar := na
        candidateEpoch := candidateEpoch + 1
        retiredStatus := ""
        color border = close < collectedOpen ? color.red : close > collectedOpen ? color.green : color.gray
        if drawReference
            h1Box := box.new(left=timestamp(tz, yy, mm, dd, 13, 0), top=h1High, right=timestamp(tz, yy, mm, dd, 22, 0), bottom=h1Low, xloc=xloc.bar_time, border_color=border, border_width=1, bgcolor=na)

previousQuote := close

// Невидимые серии нужны для цен в оповещении и автоматических проверок.
// Помимо display.none задана полная прозрачность — даже если терминал
// игнорирует режим отображения, эти серии не нарисуют диагонали.
plot(shortEvent ? eventEntry : planVisible and direction == -1 ? entry : na, "SHORT Entry", color=color.new(color.white, 100), display=display.none)
plot(shortEvent ? eventStop : planVisible and direction == -1 ? stop : na, "SHORT Stop", color=color.new(color.white, 100), display=display.none)
plot(shortEvent ? eventTarget : planVisible and direction == -1 ? point3 : na, "SHORT Target", color=color.new(color.white, 100), display=display.none)
plot(longEvent ? eventEntry : planVisible and direction == 1 ? entry : na, "LONG Entry", color=color.new(color.white, 100), display=display.none)
plot(longEvent ? eventStop : planVisible and direction == 1 ? stop : na, "LONG Stop", color=color.new(color.white, 100), display=display.none)
plot(longEvent ? eventTarget : planVisible and direction == 1 ? point3 : na, "LONG Target", color=color.new(color.white, 100), display=display.none)
plot(stage, "Stage", color=color.new(color.white, 100), display=display.data_window)
plot(direction, "Direction", color=color.new(color.white, 100), display=display.data_window)
plot(h1High, "H1 High", color=color.new(color.white, 100), display=display.data_window)
plot(h1Low, "H1 Low", color=color.new(color.white, 100), display=display.data_window)

plot(outcome, "Outcome", color=color.new(color.white, 100), display=display.data_window)
plot(showMarks and ((shortEvent and eventDraw) or (drawReference and direction == -1 and not na(signalBar) and signalBar == bar_index)) ? 1 : 0, "SHORT", color=color.new(color.white, 100), display=display.none)
